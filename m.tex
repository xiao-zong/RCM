\section{Preliminaries and notation}

\subsection*{Probability metrics and Kantorovich-Rubinstein distance}
In this paper, we investigate the closeness between two real-valued random variables, as well as between two point processes. To this end, we employ several probability distance, including the Kolmogorov distance, the Wasserstein distance, and the Kantorovich--Rubinstein distance.

The Kolmogorov distance between distributions of two random variables $X$ and $Y$ is given by 
\begin{equation*}
	\mathrm{d}_{\rm K}(X,Y):=\sup_{z\in\mathbb{R}}|\mathbb{P}(X\le z)-\mathbb{P}(Y\le z)|,
\end{equation*}
while the Wasserstein distance is 
\begin{equation*}
	\mathrm{d}_{\rm Wass}(X,Y):=\sup_{h\in\mathscr{F}_W}|\mathbb{E} h(X)-\mathbb{E} h(Y)|,
\end{equation*}
where $\mathscr{F}_W:=\{h:\mathbb{R}\to\mathbb{R}:~|h(x)-h(y)|\le |x-y|\}$.

Let $\mathbb{Y}$ be a locally compact second countable Hausdorff space with Borel $\sigma$-algebra $\mathcal{Y}$.
Let $N_{\mathbb{Y}}$ denote the space of $\sigma$-finite counting measures on $\mathbb{Y}$. Let $\mathcal{N}_{\mathbb{Y}}$ be the $\sigma$-algebra on $N_{\mathbb{Y}}$ generated by evalution mappings $\omega\mapsto\omega(A)$, $A\in\mathcal{Y}$.
A point process $\xi$ is a random element in $N_{\mathbb{Y}}$.

We define the total variation distance between two finite measures $\nu_1$ and $\nu_2$ on $\mathbb{Y}$ by 
\begin{equation*}
	\mathrm{d}_{\rm TV}(\nu_1,\nu_2):=\sup_{A\in\mathcal{Y}}|\nu_1(A)-\nu_2(A)|.
\end{equation*}
Let $\mathscr{F}_{TV}$ denote the collection of $1$-Lipschitz functions with respect to $\mathrm{d}_{\rm TV}$. The Kantorovich-Rubinstein distance between two probability measures $P,Q$ on $N_{\mathbb{Y}}$ is defined as 
\begin{equation*}
	\mathrm{d}_{\rm KR}(P,Q):= \inf_{C\in\Sigma(P,Q)}\int_{N_{\mathbb{Y}}\times N_{\mathbb{Y}}}\mathrm{d}_{\rm TV}(\omega_1,\omega_2) C(\mathrm{d}(\omega_1,\omega_2)),
\end{equation*}
where $\Sigma(P,Q)$ denotes the set of probability measures on $N_{\mathbb{Y}}\times N_{\mathbb{Y}}$ with first marginal $P$ and second marginal $Q$. 
By abuse of notation we will also write $\mathrm{d}_{\rm KR}(\xi,\zeta)$ instead of $\mathrm{d}_{\rm KR}(P,Q)$ for any two point processes $\xi,\zeta$ with distributions $P$ and $Q$ respectively.

\section{$U$-statistics of edge-marked Poisson processes}



\subsection{Edge-marked Poisson processes}\label{subsec:edge-marked_PP}

Although we are mostly interested in the random connection model, we derive general results for certain functionals and point processes constructed from so-called edge-marked Poisson processes. The idea of an edge-marked Poisson process is to take a Poisson process and to assign i.i.d.\ marks to all pairs of points. The random connection model can be derived from such an edge-marked Poisson process in the following way. We take the Poisson process of the vertices and equip each pair of vertices $x$ and $y$ with an independent random variable $U_{x,y}$ that is uniformly distributed on $[0,1]$. Then we connect $x$ and $y$ by an edge if and only if $U_{x,y}\leq \varphi(x,y)$, where $\varphi$ is the connection function.

In the following, we introduce edge-marked Poisson processes on a general measurable space. To this end let $(\mathbb{X},\mathcal{X})$ be a measurable space and let $\lambda$ be a $\sigma$-finite non-atomic measure on $\mathbb{X}$. For $\mathbb{X}$ and $\lambda$ we always assume that for each $n\in\mathbb{N}$ there exists a partition $(A_\ell^{(n)})_{\ell\in\mathbb{N}}$ of $\mathbb{X}$, i.e., $A^{(n)}_{\ell_1}\cap A^{(n)}_{\ell_2}=\emptyset$ for distinct $\ell_1,\ell_2\in\mathbb{N}$ and $\bigcup_{\ell=1}^\infty A^{(n)}_\ell=\mathbb{X}$, such that the following conditions are satisfied:
\begin{itemize}
\item For $x\in\mathbb{X}$ and $n\in\mathbb{N}$ let $A^{(n)}(x)$ be the unique $A^{(n)}_\ell$ such that $x\in A^{(n)}_\ell$. Then,
\begin{equation}\label{eqn:monotonicity_partition}
A^{(n)}(x)\subseteq A^{(m)}(x)
\end{equation}
for all $x\in\mathbb{X}$ and $m,n\in\mathbb{N}$ with $m\leq n$.
\item There exists a sequence $(\varepsilon_n)_{n\in\mathbb{N}}$ such that $\lim_{n\to\infty}\varepsilon_n=0$ and
\begin{equation}\label{eqn:size_partition}
\lambda(A_\ell^{(n)}) \leq \frac{\varepsilon_n}{\ell}
\end{equation}
for all $\ell,n\in\mathbb{N}$.
\item There exists a measurable function $\operatorname{dist}:\mathbb{X}\times \mathbb{X}\to[0,\infty)$ such that 
\begin{equation}\label{eqn:uniqueness_distances}
\lambda^3\left(\left\{(x,y,z)\in \mathbb{X}\times A^{(n)}_\ell\times A_\ell^{(n)}: \operatorname{dist}(x,y)=\operatorname{dist}(x,z)\right\}\right)=0
\end{equation}
for all $\ell,n\in\mathbb{N}$.
\end{itemize}

Let us briefly discuss choices of $\mathbb{X}$ and $\lambda$ that are relevant for studying the random connection model. For the stationary random connection model without weights one chooses $\mathbb{X}=\mathbb{R}^d$ and $\lambda=\gamma\lambda_d$, where $\lambda_d$ is the $d$-dimensional Lebesgue measure and $\gamma>0$ is an intensity parameter. For the case of stationary weighted random connection models we let $\mathbb{X}=\mathbb{R}^d\times [0,\infty)$ and $\lambda=\gamma\lambda_d\otimes \mathbb{W}$ with $\gamma$ and $\lambda_d$ as before and the weight distribution $\mathbb{W}$. In case one is only interested in the random connection model in some observation window $K$, one replaces $\mathbb{R}^d$ by $K$ and the Lebesgue measure by the restriction of the Lebesgue measure to $K$.  

Let $\eta$ be a Poisson process on $\mathbb{X}$ with the $\sigma$-finite non-atomic intensity measure $\lambda$. We assume that $\eta$ can be almost surely written as 
$$
\eta=\sum_{i\in I} \delta_{X_i}
$$
with random elements $(X_i)_{i\in I}$ on $\mathbb{X}$ and a random index set $I\subseteq\mathbb{N}$, where $\delta_x$ denotes the Dirac measure concentrated on $x$. 

Moreover, let $(\mathbb{M},\mathcal{M})$ be a measurable space with a probability measure $\mathbb{Q}$. Let $(\mathcal{M}_{i,j})_{i,j\in\mathbb{N},i\neq j}$ be a family of independent random elements on $\mathbb{M}$ distributed according to $\mathbb{Q}$. Now we assign the mark $\mathcal{M}_{i,j}$ to the pair $(X_i,X_j)$ for $i,j\in I$ with $i\neq j$. For $y,z\in\eta$ with $y\neq z$ we denote by $\mathcal{M}_{y,z}$ the mark of the pair $(y,z)$. Note that the pairs $(y,z)$ and $(z,y)$ have different marks.



\subsection{$U$-statistics and moment formulae}
For $k\in\mathbb{N}$ we use the short hand notation $[k]:=\{1,\hdots,k\}$ and define
$$
\mathbb{M}^{k\times k}_{\neq}:=\{(u_{i,j})_{i,j\in[k],i\neq j}: u_{i,j}\in \mathbb{M} \text{ for } i,j\in[k],i\neq j\}.
$$
In the following, we often denote the entries of $\mathbf{u}\in \mathbb{M}^{k\times k}_{\neq}$ by $(u_{i,j})_{i,j\in[k],i\neq j}$. Moreover let $\eta^k_{\neq}$ be the set of all $k$-tuples of distinct points of $\eta$.



For a measurable function $f: \mathbb{X}^k\times \mathbb{M}^{k\times k}_{\neq}\to\mathbb{R}$ with $k\in\mathbb{N}$ we are interested in random variables of the following form
\begin{equation}\label{def:ustat}
	S=\sum_{(x_1,\hdots,x_k)\in\eta^k_{\neq}} f(x_1,\hdots,x_k, (\mathcal{M}_{x_i,x_j})_{i,j\in[k], i\neq j}),
\end{equation}
which we call $U$-statistics of the edge-marked Poisson process. Since we sum also over all permutations of a fixed $k$-tuple of points, we can assume without loss of generality that $f$ is symmetric, i.e., 
$$
f(x_1,\hdots,x_k, (u_{i,j})_{i,j\in[k],i\neq j})=f(x_{\pi(1)},\hdots,x_{\pi(k)}, (u_{\pi(i),\pi(j)})_{i,j\in[k],i\neq j})
$$
for all $x_1,\hdots,x_k\in\mathbb{X}^k$, $\mathbf{u}\in\mathbb{M}^{k\times k}_{\neq}$ and all permutations $\pi:[k]\to[k]$. Indeed, the value of $S$ does not change if we replace $f$ by its symmetrisation
$$
\widetilde{f}(x_1,\hdots,x_k, (u_{i,j})_{i,j\in[k],i\neq j}):=\frac{1}{k!}\sum_{\pi\in \mathrm{P}([k])}f(x_{\pi(1)},\hdots,x_{\pi(k)}, (u_{\pi(i),\pi(j)})_{i,j\in[k],i\neq j})
$$
for $x_1,\hdots,x_k\in\mathbb{X}^k$ and $\mathbf{u}\in\mathbb{M}^{k\times k}_{\neq}$, where $\mathrm{P}([k])$ is the set of all permutations of $[k]$.
We often formulate assumptions on terms of the $U$-statistic
\begin{equation}\label{def-integ1}
	\overline{S}:=\sum_{(x_1,\hdots,x_k)\in\eta^k_{\neq}} \left|f(x_1,\hdots,x_k, (\mathcal{M}_{x_i,x_j})_{i,j\in[k], i\neq j})\right|.
\end{equation}
Note that $S=\overline{S}$ if $f$ is non-negative.

We start with formulae for expectations and variances of $U$-statistics of edge-marked Poisson processes. From now on we always assume that the function $f$ is symmetric as discussed above.

\begin{prop}\label{prop:expectation_variance}
    % Let $f: \mathbb{X}^k\times \mathbb{M}^{k\times k}_{\neq}\to\mathbb{R}$ with $k\in\mathbb{N}$ be a measurable function. 
    \begin{enumerate}
    \item[]
    \item[a)]
    Suppose $\E(\overline{S})<\infty$, then we have 
\begin{equation}\label{eqn:generalised_Mecke_U-statistics}
\E S = \int_{\mathbb{X}^k}\int_{\mathbb{M}^{k\times k}_{\neq}} f(x_1,\hdots,x_k,\mathbf{u}) \, \mathbb{Q}_{\neq}^{k\times k}(\mathrm{d}\mathbf{u}) \, \lambda^k(\mathrm{d}(x_1,\hdots,x_k)).
\end{equation}
\item[b)] \textcolor{red}{Suppose $\E\left(\overline{S}^2\right)<\infty$, then we have} 
\begin{align}\label{eqn:variance}
\Var S =\sum_{\ell=1}^k{k\choose \ell}^2\ell! &\int_{\mathbb{X}^{2k-\ell}}\int_{\mathbb{M}^{(2k-\ell)\times (2k-\ell)}_{\neq}} f(x_1,\hdots,x_k,(u_{i,j})_{i,j\in[k],i\ne j}) \nonumber\\
&f(x_1,\dots,x_{\ell},x_{k+1},\dots,x_{2k-\ell},(u_{i,j})_{i,j\in[\ell]\cup\{k+1,\dots,2k-\ell\},i\ne j}) )\\
&\mathbb{Q}_{\neq}^{(2k-\ell)\times (2k-\ell)}(\mathrm{d}\mathbf{u}) \, \lambda^{2k-\ell}(\mathrm{d}(x_1,\hdots,x_{2k-\ell})).\nonumber
\end{align}
\end{enumerate}
\end{prop}

\subsection{Poisson process approximation}

For $k\in\mathbb{N}$ with $k\geq2$ and a locally compact second countable Hausdorff space $\mathbb{Y}$ with Borel $\sigma$-algebra $\mathcal{Y}$ we consider a measurable symmetric function $g: \operatorname{dom} g\to\mathbb{Y}$ such that $\operatorname{dom} g\subseteq \mathbb{X}^k\times \mathbb{M}^{k\times k}_{\neq}$ is also measurable. Moreover, we define
$$
{\operatorname{dom}'} g := \left\{( x_1, \ldots, x_k)\in\mathbb{X}^k : \text{ there exists a } \mathbf{u}\in\mathbb{M}^{k\times k}_{\neq} \text{ such that } \left( x_1, \ldots, x_k, \mathbf{u} \right) \in  \operatorname{dom} g\right\}.
$$
In the following, we consider the point process
\begin{equation}\label{definition_xi}
	\xi:= \frac{1}{k!}\sum_{(x_1, \ldots, x_k) \in \eta^k_{\neq}\cap  {\operatorname{dom}'} g}  \delta_{g \left( x_1, \ldots, x_k, (\mathcal{M}_{x_i,x_j})_{i,j\in[k], i \neq j} \right)}.
\end{equation}
The intensity measure $\mathbf{L}$ of $\xi$ is given by
$$
\mathbf{L}(A):= \mathbb{E} \xi(A) =\frac{1}{k!} \int_{\operatorname{dom} g} \mathbf{1}\{ g(x_1,\hdots,x_k,\mathbf{u})\in A \} \, \lambda^k\otimes \mathbb{Q}_{\neq}^{k\times k}(\mathrm{d} (x_1,\hdots,x_k,\mathbf{u})) 
$$
for $A\in\mathcal{Y}$.

\begin{theorem} \label{Poisson_process_approximation}
Let $\xi$ be the point process given in \eqref{definition_xi} and assume that $\E \xi(\mathbb{Y})^2<\infty$. Let $\zeta$ be a Poisson process with finite intensity measure $\mathbf{M}$. Then,
    \begin{align*}
        \mathrm{d}_{\rm KR}({\xi}, \zeta) & \leq \mathrm{d}_{\mathrm{TV}}(\mathbf{L},\mathbf{M}) + 2 \big( \E \xi(\mathbb{Y})^2 - (\E \xi(\mathbb{Y}))^2 - \E \xi(\mathbb{Y}) \big) \\
        & \leq \mathrm{d}_{\mathrm{TV}}(\mathbf{L},\mathbf{M}) + \frac{2^{k+1}}{k!} \, r(\operatorname{dom} g),
    \end{align*}
where
\begin{align*}
  &  r(\operatorname{dom} g) := \max_{1 \leq \ell \leq k-1} 
\int_{\mathbb{X}^{2k-\ell}} \int_{\mathbb{M}^{(2k-\ell)\times (2k-\ell)}_{\neq}}  
\mathbf{1}\big\{ ( {x}_1, \ldots, {x}_k , (u_{i,j})_{i,j\in[k],i\neq j} ) \in \operatorname{dom} g  \big\}  \\
&  \hspace{3cm} \times \mathbf{1}\big\{ ( {x}_1, \ldots, {x}_\ell, {x}_{k+1}, \ldots, {x}_{2k-\ell}, (u_{i,j})_{i,j\in [\ell]\cup\{k+1,\hdots,2k-\ell\}, i\neq j} ) \in \operatorname{dom} g \big\} \\
& \hspace{3cm} \times  \mathbb{Q}^{(2k-\ell)\times(2k-\ell)}_{\neq} (\mathrm{d}\mathbf{u}) \, \lambda^{2k-\ell}(\mathrm{d}(x_1,\hdots,x_{2k-\ell})).
\end{align*}  
\end{theorem}

Next we provide some corollaries of Theorem \ref{Poisson_process_approximation}. For $g: \operatorname{dom} g \to\mathbb{Y}$ as above and $B\in\mathcal{Y}$ we define the $U$-statistic
$$
T:=\frac{1}{k!}\sum_{(x_1,\hdots,x_k)\in\eta^k_{\neq}\cap \operatorname{dom}'g} \mathbf{1}\{g(x_1,\hdots,x_k,(\mathcal{M}_{x_i,x_j})_{i,j\in[k],i\neq j})\in B\}.
$$

\begin{cor}\label{cor:Poisson_approximation}
Assume that $\mathbb{E}[T^2]<\infty$ and let $Z$ be a Poisson distributed random variable with parameter $a\in[0,\infty)$. Then,
\begin{align*}
	\mathrm{d}_{\rm Wass}(T,Z) & \leq |\mathbb{E}[T]-a| + 2 ( \mathbb{E}[T^2]- \mathbb{E}[T] -(\mathbb{E}[T])^2 ) \\
& \leq |\mathbb{E}[T]-a| + \frac{2^{k+1}}{k!} r(B)
\end{align*}
with
\begin{align*}
  &  r(B) := \max_{1 \leq \ell \leq k-1} 
\int_{\mathbb{X}^{2k-\ell}} \int_{\mathbb{M}^{(2k-\ell)\times (2k-\ell)}_{\neq}}  
\mathbf{1}\big\{ g({x}_1, \ldots, {x}_k , (u_{i,j})_{i,j\in[k],i\neq j} ) \in B \big\}  \\
&  \hspace{3cm} \times \mathbf{1}\big\{ g({x}_1, \ldots, {x}_\ell, {x}_{k+1}, \ldots, {x}_{2k-\ell}, (u_{i,j})_{i,j\in [\ell]\cup\{k+1,\hdots,2k-\ell\}, i\neq j} ) \in B \big\} \\
& \hspace{3cm} \times  \mathbb{Q}^{(2k-\ell)\times(2k-\ell)}_{\neq} (\mathrm{d}\mathbf{u}) \, \lambda^{2k-\ell}(\mathrm{d}(x_1,\hdots,x_{2k-\ell})).
\end{align*}  
\end{cor}

\begin{proof}
The statement follows from Theorem \ref{Poisson_process_approximation} exactly as Theorem 7.1 follows from Theorem 3.1 in \cite{dst}.
\end{proof}

Let $g: \mathbb{X}^k\times\mathbb{M}^{k\times k}_{\neq}\to\mathbb{R}$ be a measurable symmetric function. Define the $U$-statistic
\begin{equation}\label{definition_U}
	U := \frac{1}{k!}\sum_{(x_1,\hdots,x_k)\in\eta^k_{\neq}} g(x_1,\hdots,x_k,(\mathcal{M}_{x_i,x_j})_{i,j\in[k],i\neq j}). 
\end{equation}
Moreover, let
\begin{align*}
\mathbf{L}(A) := \frac{1}{k!} \, \mathbb{E} \sum_{(x_1, \ldots, x_k) \in \eta_{\neq}^k} 
\mathbf{1}\left\{ g(x_1, \ldots, x_k, (\mathcal{M}_{x_i,x_j})_{i,j\in[k], i \neq j}) \in A \setminus \{0\} \right\}
\end{align*}
for $A \in \mathcal{B}(\mathbb{R})$, where the second equality follows from Proposition \ref{prop:expectation_variance}. We get the following result.

\begin{cor}\label{compound}
For a Poisson process  \( \zeta \) on \( \mathbb{R} \) with a finite intensity measure \( \mathbf{M} \) let
\[
Y := \sum_{x \in \zeta} x.
\]
Then,
\begin{align*}
\mathrm{d}_{\mathrm{TV}}(U, Y) \le \mathrm{d}_{\mathrm{TV}}(\mathbf{L}, \mathbf{M}) + \frac{2^{k+1}}{k!} \, r_0
\end{align*}
with
\begin{align*}
  &  r_0 := \max_{1 \leq \ell \leq k-1} 
\int_{\mathbb{X}^{2k-\ell}} \int_{\mathbb{M}^{(2k-\ell)\times (2k-\ell)}_{\neq}}  
\mathbf{1}\big\{ g({x}_1, \ldots, {x}_k , (u_{i,j})_{i,j\in[k],i\neq j} )\neq 0 \big\}  \\
&  \hspace{3cm} \times \mathbf{1}\big\{ g({x}_1, \ldots, {x}_\ell, {x}_{k+1}, \ldots, {x}_{2k-\ell}, (u_{i,j})_{i,j\in [\ell]\cup\{k+1,\hdots,2k-\ell\}, i\neq j} ) \neq 0 \big\} \\
& \hspace{3cm} \times  \mathbb{Q}^{(2k-\ell)\times(2k-\ell)}_{\neq} (\mathrm{d}\mathbf{u}) \, \lambda^{2k-\ell}(\mathrm{d}(x_1,\hdots,x_{2k-\ell})).
\end{align*}
\end{cor}

\begin{proof}
The assertion can be deduced from Theorem \ref{Poisson_process_approximation} as Theorem 7.4 is derived from Theorem 3.1 in \cite{dst}.
\end{proof}

Next we compare the distribution of the random variable $U$ introduced in \eqref{definition_U} with an $\alpha$-stable distribution. To this end we fix $\alpha\in(0,1)$ and let
$$
g_\alpha (\mathbf{x}_m) :=  \operatorname{sign}(g(\mathbf{x}_m)) g(\mathbf{x}_m)^{-\alpha}
$$
for $\mathbf{x}_m\in \mathbb{X}^k\times\mathbb{M}^{k\times k}_{\neq}$, where $\operatorname{sign}(u) = \mathbf{1}\{u \ge 0\} - \mathbf{1}\{u < 0\}$ for $u\in\mathbb{R}$.
We define
\begin{align*}
	\mathbf{L}^{(\alpha)}(A) & := \frac{1}{k!} \, \mathbb{E} \sum_{(x_1, \ldots, x_k) \in \eta_{\neq}^k} \mathbf{1}\{ g(\mathbf{x}_m)\neq 0 \}  \mathbf{1}\{ g_\alpha(x_1, \ldots, x_k, (\mathcal{M}_{x_i,x_j})_{i,j\in[k], i \neq j})\in A \}
\end{align*}
for $A \in \mathcal{B}(\mathbb{R})$.

\begin{cor}\label{approximation_alpha-stable}
Let \( \alpha \in (0, 1) \), and let \( \mathbf{M} \) be either the Lebesgue measure on \( \mathbb{R} \) or its restriction to \( (0,\infty) \). Define
\[
Y := \sum_{x \in \zeta} \operatorname{sign}(x) |x|^{-1/\alpha},
\]
where \( \zeta \) is a Poisson process with intensity measure \( \mathbf{M}\). Assume that there exist functions \( g_1, g_2, g_3 \colon (0,\infty) \to [0,\infty) \) such that, for every \( a > 0 \),
\begin{align*}
\mathrm{d}_{\mathrm{TV}}(\mathbf{L}^{(\alpha)}|_{[-a,a]}, \mathbf{M}|_{[-a,a]}) &\le g_1(a),
\end{align*}
\begin{align*}
&\max_{1 \leq \ell \leq k-1} 
\int_{\mathbb{X}^{2k-\ell}} \int_{\mathbb{M}^{(2k-\ell)\times (2k-\ell)}_{\neq}}  
\mathbf{1}\big\{ g({x}_1, \ldots, {x}_k , (u_{i,j})_{i,j\in[k],i\neq j} )\neq 0 \big\}\mathbf{1}\big\{ g_\alpha({x}_1, \ldots, {x}_k , (u_{i,j})_{i,j\in[k],i\neq j} )\in [-a,a] \big\}  \\
&  \hspace{3cm} \times  \mathbf{1}\big\{ g({x}_1, \ldots, {x}_\ell, {x}_{k+1}, \ldots, {x}_{2k-\ell}, (u_{i,j})_{i,j\in [\ell]\cup\{k+1,\hdots,2k-\ell\}, i\neq j} ) \neq 0 \big\}\\
& \hspace{3cm} \times \mathbf{1}\big\{ g_\alpha({x}_1, \ldots, {x}_\ell, {x}_{k+1}, \ldots, {x}_{2k-\ell}, (u_{i,j})_{i,j\in [\ell]\cup\{k+1,\hdots,2k-\ell\}, i\neq j} ) \in [-a,a] \big\}\\
& \hspace{3cm} \times  \mathbb{Q}^{(2k-\ell)\times(2k-\ell)}_{\neq} (\mathrm{d}\mathbf{u}) \, \lambda^{2k-\ell}(\mathrm{d}(x_1,\hdots,x_{2k-\ell}))\\
&\le g_2(a),
\end{align*}
\begin{align*}
& \frac{1}{k!} \, \mathbb{E} \sum_{(x_1, \ldots, x_k) \in \eta^k_{\neq}} \mathbf{1}\{|g(x_1, \ldots, x_k,(\mathcal{M}_{x_i,x_j})_{i,j\in[k], i \neq j})| < a^{-1/\alpha}\} |g(x_1, \ldots, x_k,(\mathcal{M}_{x_i,x_j})_{i,j \in[k], i \neq j})| \\
&\le g_3(a).
\end{align*}
Then there exists a constant \( C > 0 \), depending only on \( \alpha \) and \( k \), such that
\[
d_{\operatorname{Kol}}(U, Y) \le C b,
\]
where
\[
b := \inf_{a > 0} \max\left\{ a^{1/2 - 1/(2\alpha)},\; g_1(a),\; g_2(a),\; \sqrt{g_3(a)} \right\}.
\]
\end{cor}

\begin{proof}
The proof is analogously to the derivation of Theorem 7.8 from Theorem 3.1 in \cite{dst}. 
\end{proof}

\begin{align*}
  &  \int_{\hat{X}^\ell}  
\int_{\hat{X}^{k - \ell}}\int_{\hat{X}^{k - \ell}} 
\mathbf{1}\big( \hat{f}_t ( \hat{x}_1, \ldots, \hat{x}_k ) \in B \big) \,  \mathbf{1}\big( \hat{f}_t ( \hat{x}_1, \ldots, \hat{x}_\ell, \hat{x}'_{\ell+1} \ldots, \hat{x}'_{k} ) \in B \big) \\
& \qquad \times  \mathbf{1}\big(  ( {x}_1, \ldots, {x}_k, {x}'_{\ell+1} \ldots, {x}'_{k} ) \notin \Delta_{2k-\ell} \big) \\
& \qquad \times \hat{K}_t^{k - \ell}(d(\hat{x}_{\ell+1}, \ldots, \hat{x}_k)) \hat{K}_t^{k - \ell}(d(\hat{x}'_{\ell+1}, \ldots, \hat{x}'_k))\hat{K}_t^\ell(d(\hat{x}_1, \ldots, \hat{x}_\ell)) \\
& = \mathbb{E} \Bigg[ \int_{X^\ell}  
\int_{X^{k - \ell}}\int_{X^{k - \ell}} 
\mathbf{1}\big(  ( {x}_1, \ldots, {x}_k, {x}'_{\ell+1} \ldots, {x}'_{k} ) \notin \Delta_{2k-\ell} \big) \,  \\
& \qquad \times \mathbf{1}\big( {f}_t ( {x}_1, \ldots, {x}_k , W_{x_i,x_j} ) \in B \big)\mathbf{1}\big( {f}_t ( {x}_1, \ldots, {x}_\ell, {x}'_{\ell+1} \ldots, {x}'_{k}, \tilde{W}_{i,j} ) \in B \big) \\
& \qquad \times {K}_t^{k - \ell}(d({x}_{\ell+1}, \ldots, {x}_k)) {K}_t^{k - \ell}(d({x}'_{\ell+1}, \ldots, {x}'_k)) {K}_t^\ell(d({x}_1, \ldots, {x}_\ell)) \Bigg] ,
\end{align*}
where now $K_t = tK$ denotes the intensity measure of $\eta$, and the random variables $\tilde{W}_{i,j}$ above are taken to be ${W}_{x_i,x_j}$ if $i,j \in [\ell]^2$ and $\overline{W}_{i,j}$ if $i,j \in [k]^2 \setminus [\ell]^2$ with $\overline{W}_{i,j}$ being an independent copy of the weight(s). Note that if $\mathbb{Q}$ denotes the distribution of the weights, we can write

\begin{align*}
   & \mathbb{E} \Bigg[ \int_{X^\ell}  
\int_{X^{k - \ell}}\int_{X^{k - \ell}} 
\mathbf{1}\big(  ( {x}_1, \ldots, {x}_k, {x}'_{\ell+1} \ldots, {x}'_{k} ) \notin \Delta_{2k-\ell} \big) \,  \\
& \qquad \times \mathbf{1}\big( {f}_t ( {x}_1, \ldots, {x}_k , W_{x_i,x_j} ) \in B \big)\mathbf{1}\big( {f}_t ( {x}_1, \ldots, {x}_\ell, {x}'_{\ell+1} \ldots, {x}'_{k}, \tilde{W}_{i,j} ) \in B \big) \\
& \qquad \times {K}_t^{k - \ell}(d({x}_{\ell+1}, \ldots, {x}_k)) {K}_t^{k - \ell}(d({x}'_{\ell+1}, \ldots, {x}'_k)) {K}_t^\ell(d({x}_1, \ldots, {x}_\ell)) \Bigg] \\
& = \int \int_{X^\ell}  
\int_{X^{k - \ell}}\int_{X^{k - \ell}} 
\mathbf{1}\big(  ( {x}_1, \ldots, {x}_k, {x}'_{\ell+1} \ldots, {x}'_{k} ) \notin \Delta_{2k-\ell} \big) \,  \\
&  \times \mathbf{1}\big( {f}_t ( {x}_1, \ldots, {x}_k , w ) \in B \big)\mathbf{1}\big( {f}_t ( {x}_1, \ldots, {x}_\ell, {x}'_{\ell+1} \ldots, {x}'_{k}, w ) \in B \big) \\
&  \times {K}_t^{k - \ell}(d({x}_{\ell+1}, \ldots, {x}_k)) {K}_t^{k - \ell}(d({x}'_{\ell+1}, \ldots, {x}'_k)) {K}_t^\ell(d({x}_1, \ldots, {x}_\ell)) d\mathbb{Q}^{\frac{(2k - \ell) (2k -\ell-1)}{2}} (dw).
\end{align*}

We obtain the following result regarding Poisson approximation for the original U-statistics.

\begin{theorem} \label{P1}
    Let $S$ be the U-statistics given in \eqref{originalUstat}. Let $Z$ be a Poisson distributed random variable with mean $\lambda$. Assume either that
    \begin{itemize}
        \item[(i)] the intensity measure $K_t \otimes \mathbb{Q}$ is finite
    \end{itemize}
    or
    \begin{itemize}
        \item[(ii)] $\mathbb{P} (\exists r_0>0 \,: \, \hat{\eta}( \mathbb{R}^d \times \cdot) = \hat{\eta}( (B(0,r) \times \cdot) \, \forall r \geq r_0 ) = 1$.
    \end{itemize}
    Then it holds
    \begin{align*}
	    \mathrm{d}_{\mathrm{Wass}}({S}, Z) \leq \left| \mathbb{E}[{S}] - \lambda \right| + \frac{2^{k+1}}{k!} \, s_t(B)
    \end{align*}
for
\begin{align*}
  &  s_t(B) := \max_{1 \leq \ell \leq k-1} 
\int \int_{X^\ell}  
\int_{X^{k - \ell}}\int_{X^{k - \ell}} \mathbf{1}\big( {f}_t ( {x}_1, \ldots, {x}_k , w ) \in B \big)  \\
&  \times \mathbf{1}\big( {f}_t ( {x}_1, \ldots, {x}_\ell, {x}'_{\ell+1} \ldots, {x}'_{k}, w ) \in B \big){K}_t^{k - \ell}(d({x}_{\ell+1}, \ldots, {x}_k)) \\
&  \times  {K}_t^{k - \ell}(d({x}'_{\ell+1}, \ldots, {x}'_k)) {K}_t^\ell(d({x}_1, \ldots, {x}_\ell)) d\mathbb{Q}^{\frac{(2k - \ell) (2k -\ell-1)}{2}} (dw) .
\end{align*}
  
\end{theorem}




\subsubsection{Compound Poisson approximation}

As another consequence of Theorem \ref{Poisson_process_approximation} we get the analogue of \cite[Theorem 7.4]{dst}. Consider the family of U-statistics

\begin{align*}
S_t := \frac{1}{k!} \sum_{(x_1, \ldots, x_k) \in \eta_{t,\neq}^k} h_t(x_1, \ldots, x_k, (W_{x_i,x_j})_{i,j =1,\ldots, k, i \neq j}), \quad t \ge 1,
\end{align*}
for a symmetric function $h_t$. Then define

\begin{align*}
L_t(A) := \frac{1}{k!} \, \mathbb{E} \sum_{(x_1, \ldots, x_k) \in \eta_{t, \neq}^k} 
\mathbf{1}\left( h_t(x_1, \ldots, x_k, (W_{x_i,x_j})_{i,j =1,\ldots, k, i \neq j}) \in t^{-\gamma} A \setminus \{0\} \right),
\end{align*}
for $A \in \mathcal{B}(\mathbb{R})$. We get the following result.

\begin{theorem}\label{compound}
Let \( \zeta \) be a Poisson process on \( \mathbb{R} \) with a finite intensity measure \( M \), let
\[
Z := \sum_{x \in \zeta} x,
\]
and let \( \gamma \in \mathbb{R} \). Then, for \( t \ge 1 \),
\begin{align*}
\mathrm{d}_{\mathrm{TV}}(t^{\gamma} S_t, Z) \le \mathrm{d}_{\mathrm{TV}}(L_t, M) + \frac{2^{k+1}}{k!} \, r_t,
\end{align*}
with
\begin{align*}
  &  r_t(A) := \max_{1 \leq \ell \leq k-1} 
\int \int_{X^\ell}  
\int_{X^{k - \ell}}\int_{X^{k - \ell}} \\
& \, \mathbf{1}\big(  h_t( {x}_1, \ldots, {x}_k , w ) \neq 0  \big) \mathbf{1}\big( h_t( {x}_1, \ldots, {x}_\ell, {x}'_{\ell+1} \ldots, {x}'_{k}, w ) \neq 0 \big) \\
&\,{K}^{k - \ell}(d({x}_{\ell+1}, \ldots, {x}_k))   {K}^{k - \ell}(d({x}'_{\ell+1}, \ldots, {x}'_k)) {K}^\ell(d({x}_1, \ldots, {x}_\ell)) d\mathbb{Q}^{\frac{(2k - \ell) (2k -\ell-1)}{2}} (dw) .
\end{align*}
\end{theorem}








\subsubsection{Approximation by $\alpha$-stable distribution}

Fix $\alpha \in (0, 1)$ and $\gamma \in \mathbb{R}$. The goal is to make use of a functional limit theorem for the point processes
\begin{align*}
    t^\gamma \cdot \xi_t &:= \frac{1}{k!} \sum_{(x_1, \ldots, x_k) \in \eta^k_{\neq}} 
\mathbf{1}(h_t(x_1, \ldots, x_k,(W_{x_i,x_j})_{i,j =1,\ldots, k, i \neq j}) \ne 0) \\
&\quad \delta_{\operatorname{sign}(h_t(x_1, \ldots, x_k,(W_{x_i,x_j})_{i,j =1,\ldots, k, i \neq j})) t^\gamma |h_t(x_1, \ldots, x_k,(W_{x_i,x_j})_{i,j =1,\ldots, k, i \neq j})|^{-\alpha}},
\end{align*}
 $t \ge 1$, on $\mathbb{R}$, with $\operatorname{sign}(a) = \mathbf{1}(a \ge 0) - \mathbf{1}(a < 0)$. In fact, having established Theorem~\ref{Poisson_process_approximation} the following application follows from the proof of \cite[Theorem 7.8]{dst}. In the following we denote by $L_t$ the intensity measure of $t^\gamma \cdot \xi_t$, and

\begin{align*}
  &  r_t(A) := \max_{1 \leq \ell \leq k-1} 
\int \int_{X^\ell}  
\int_{X^{k - \ell}}\int_{X^{k - \ell}} \\
& \quad \mathbf{1}\big(  h_t( {x}_1, \ldots, {x}_k , w ) \neq 0  \big) \mathbf{1}\left( \operatorname{sign}(h_t(x_1, \ldots, x_k,w)) \, |h_t(x_1, \ldots, x_k,w)|^{-\alpha} \in t^{-\gamma} A \right)
 \\
&  \times \mathbf{1}\big( h_t( {x}_1, \ldots, {x}_\ell, {x}'_{\ell+1} \ldots, {x}'_{k}, w ) \neq 0 \big) \\
& \times \mathbf{1}\left( \operatorname{sign}(h_t({x}_1, \ldots, {x}_\ell, {x}'_{\ell+1} \ldots, {x}'_{k},w)) \, |h_t({x}_1, \ldots, {x}_\ell, {x}'_{\ell+1} \ldots, {x}'_{k},w)|^{-\alpha} \in t^{-\gamma} A \right)\\
&\,{K}^{k - \ell}(d({x}_{\ell+1}, \ldots, {x}_k))   {K}^{k - \ell}(d({x}'_{\ell+1}, \ldots, {x}'_k)) {K}^\ell(d({x}_1, \ldots, {x}_\ell)) d\mathbb{Q}^{\frac{(2k - \ell) (2k -\ell-1)}{2}} (dw) .
\end{align*}

\begin{theorem}\label{stable}
Let \( \alpha \in (0, 1) \), and let \( M \) be either the Lebesgue measure on \( \mathbb{R} \) or its restriction to \( \mathbb{R}_+ \). Define
\[
Z := \sum_{x \in \zeta} \operatorname{sign}(x) |x|^{-1/\alpha},
\]
where \( \zeta \) is a Poisson process with intensity measure \( M\). Assume that there exist a constant \( \gamma \in \mathbb{R} \) and functions \( g_1, g_2, g_3 \colon \mathbb{R}_+^2 \to \mathbb{R}_+ \) such that, for every \( a > 0 \) and \( t \ge 1 \),
\begin{align*}
\mathrm{d}_{\mathrm{TV}}(L_t|_{[-a,a]}, M|_{[-a,a]}) &\le g_1(a, t), \\
r_t([-a,a]) &\le g_2(a, t), \\
\frac{t^{-\gamma/\alpha}}{k!} \, \mathbb{E} \sum_{(x_1, \ldots, x_k) \in \eta^k_{\neq}}
\frac{\mathbf{1}(|h_t(x_1, \ldots, x_k,(W_{x_i,x_j})_{i,j =1,\ldots, k, i \neq j})| < t^{\gamma/\alpha} a^{-1/\alpha})}{|h_t(x_1, \ldots, x_k,(W_{x_i,x_j})_{i,j =1,\ldots, k, i \neq j})|}
&\le g_3(a, t).
\end{align*}
Then there exists a constant \( C > 0 \), depending only on \( \alpha \) and \( k \), such that
\[
	\mathrm{d}_{\mathrm{K}}(t^{-\gamma/\alpha} S_t, Z) \le C g(t), \quad t \ge 1,
\]
where
\[
g(t) := \inf_{a > 0} \max\left\{ a^{1/2 - 1/(2\alpha)},\; g_1(a,t),\; g_2(a,t),\; \sqrt{g_3(a,t)} \right\}.
\]
\end{theorem}

\subsection{subgraph count}
Let $G=(V_G,E_G)$ be a fixed connected graph of order $k$, $k\ge2$, with vertex set $V_G$ and edge set $E_G$. 
%For any two graphs $G_1$ and $G_2$, we denote $G_1\simeq G_2$ when $G_1$ is isomorphic to $G_2$. 
Denote $|\mathrm{Aut}(G)|$ as the number of automorphism of $G$. 
To ensure normal approximation, we need the following minimal condition for the connection function $\varphi$.
\begin{ass}\label{graph-ass1}
	Let $G$ be a connected graph with $k\ge2$ vertices.
\begin{enumerate}
	\item[i)] 
\begin{equation}
	\mathbb{P}(\exists\ H\subseteq \mathbb{G}_{\varphi}(\eta):\ H\simeq G)>0.
\end{equation}
\item[ii)]
\begin{equation}
	\mathbb{P}\left(\exists\ \{x_1,\dots,x_k\}\subseteq\eta:\ \mathbb{G}_{\varphi}(\{x_1,\dots,x_k\})\simeq G\right)>0.
\end{equation}
\end{enumerate}
\end{ass}
\begin{theorem}\label{propgraph}
Let $G$ be a connected graph with $k\ge2$ vertices.
Denote by $N_G^{\subset}$ and $N_G^{=}$ the non-induced subgraph count and induced subgraph counts, respectively, of copies of $G$ in 
$\mathbb{G}_{\varphi}(\eta\cap\Gamma_s)$, defined by
\begin{align}
	N_G^{\subset}&:=\frac1{|\mathrm{Aut}(G)|}\sum_{(x_1,\dots,x_k)\in\eta^k_{\ne}\cap\Gamma_s}\prod_{\{i,j\}\in E_G}\mathbf{1}\{x_i\leftrightarrow x_j\},\label{def:subgraph-count1}\\
	N_G^{=}&:=\frac1{|\mathrm{Aut}(G)|}\sum_{(x_1,\dots,x_k)\in\eta^k_{\ne}\cap\Gamma_s}\prod_{\{i,j\}\in E_G}
    \mathbf{1}\{x_i\leftrightarrow x_j\}\prod_{\{\imath,\jmath\}\notin E_G}\mathbf{1}\{x_{\imath}\nleftrightarrow x_{\jmath}\}.\label{def:subgraph-count2}
\end{align}
If Assumption~\ref{graph-ass1}-i) holds, then there exist constants $C,s_0>0$, depending only on $\varphi$, $k$ and $G$ such that when $s\ge s_0$,
\begin{equation}
	\mathrm{d}_{\mathrm{Wass}}\left(\frac{N_G^{\subset}-\mathbb{E} N_G^{\subset}}{\sqrt{\Var N_G^{\subset}}},N\right)\le \mathrm{d}_{\mathrm{K}}\left(\frac{N_G^{\subset}-\mathbb{E} N_G^{\subset}}{\sqrt{\Var N_G^{\subset}}},N\right)\le\frac{C}{\sqrt{s}}.
\end{equation}
Moreover, if Assumption~\ref{graph-ass1}-ii) holds, the same bound applies to the induced subgraph count $N_G^{=}$.
\end{theorem}
Before proving Theorem~\ref{propgraph}, we first show that as a direct consequence of Proposition~\ref{prop:moments}, the growth of cumulants of subgraph counts is well controlled. 
\begin{prop}
For any $\ell\in\mathbb{N}$ and $i\in\{=,\subset\}$, we have, for each $\pi\in\widetilde{\Pi}^\ell(k)$
\begin{equation}\label{intgrand-0}
\int_{\Gamma_s^{|\pi|}}\int_{[0,1]^{|\pi|\times|\pi|}_{\ne}}((f^{(i)}_G)^{\otimes\ell})_{\pi}(\bm{x},\bm{u})\mathrm{d}\bm{u}\lambda_d^{|\pi|}(\mathrm{d}\bm{x})\le

\end{equation}
Moreover, we have 
\begin{equation}
	\cum_\ell(N_G^{(i)})\le \frac{1}{|\mathrm{Aut}(G)|^\ell}\left[\left(\gamma \int_{\mathbb{R}^d}\varphi(z)\mathrm{d}z\right)\vee1\right]^{\ell(k-1)+1}k^{k\ell}(\ell!)^k s.
\end{equation}
 \end{prop}

Define $f^{\subset}_G:(\R^d)^k\times [0,1]^{k(k-1)/2}\to\{0,1\}$ as 
\begin{equation}
	f^{\subset}_G(x_1,\dots,x_k,\mathbf{u}):=\frac1{|\mathrm{Aut}(G)|}\prod_{\{i,j\}\in E_G}\mathbf{1}\{u_{i,j}\le\varphi(x_i-x_j)\},
\end{equation}
with $\mathbf{u}=(u_{i,j})_{1\le i<j\le k}$ and $u_{i,j}:=u_{j,i}$ if $i>j$. 
Similarly, we define 
$f^{=}_G:(\R^d)^k\times [0,1]^{k(k-1)/2}\to\{0,1\}$ as 
\begin{equation}
	f^{=}_G(x_1,\dots,x_k,\mathbf{u}):=\frac1{|\mathrm{Aut}(G)|}\prod_{\{i,j\}\in E_G}\mathbf{1}\{u_{i,j}\le\varphi(x_i-x_j)\}\prod_{\{\imath,\jmath\}\notin E_G}\mathbf{1}\{u_{\imath,\jmath}>\varphi(x_{\imath}-x_{\jmath})\}.
\end{equation}
	Because of \eqref{momeq-3}, it is necessary to show that for each $\pi\in\widetilde{\Pi}^\ell(k)$
	\begin{equation}\label{intgrand-1}
		\int_{\Gamma_s^{|\pi|}}\int_{[0,1]^{|\pi|\times|\pi|}_{\ne}}((f^{(i)}_G)^{\otimes\ell})_{\pi}(\bm{x},\bm{u})\mathrm{d}\bm{u}\lambda_d^{|\pi|}(\mathrm{d}\bm{x})\le C s
\end{equation}
For $n\in\mathbb{N}$, we use the abbreviation $\mathbf{I}_n:=\{(i,j)\in[n]^2:i<j\}$ and denote

\end{equation*}
Because graph $G$ is connected, for any $\pi\in\widetilde{\Pi}^2(\ell)$, the graph with vertex set 
$$
\left\{x_i^{(1)},x_i^{(2)},\dots,x_i^{(\ell)}:i\in[k]\right\}_{\pi},
$$
and edge set 
$$

is a connected graph with $|\pi|$ vertices, denoted by $(G^{\otimes \ell})_{|\pi|}$.
Consequently, there exists a spanning tree of $(G^{\otimes \ell})_{|\pi|}$, which is isomorphic to a tree $I_{|\pi|}\in\mathcal{I}_{|\pi|}$.

\begin{align}

&=\frac{1}{|\mathrm{Aut}(G)|^\ell}\mathrm{Vol}(\Gamma_s)\left(\int_{\mathbb{R}^d}\varphi(z)\mathrm{d}z\right)^{|\pi|-1}\gamma^{|\pi|},\label{intgrand-2}
\end{align}
where the last equation is obtained by integrating successively beginning with the variables, except $x_1$, whose indices are leaves of the tree $I_{|\pi|}$.
Because $|\pi|\le \ell(k-1)+1$ for all $\pi\in\widetilde{\Pi}^\ell(k)$, combining \eqref{intgrand-2} with cumulant identities in \eqref{momeq-3}, we have 
\begin{align}
    	\cum_\ell(N_G^{I})&\le \sum_{\pi\in\widetilde{\Pi}^\ell(k)}\frac{1}{|\mathrm{Aut}(G)|^\ell}\mathrm{Vol}(\Gamma_s)\left(\int_{\mathbb{R}^d}\varphi(z)\mathrm{d}z\right)^{|\pi|-1}\gamma^{|\pi|}\nonumber\\
        &\le \frac{1}{|\mathrm{Aut}(G)|^\ell}\left[\left(\gamma \int_{\mathbb{R}^d}\varphi(z)\mathrm{d}z\right)\vee1\right]^{\ell(k-1)+1}\left|\widetilde{\Pi}^\ell(k)\right|s\nonumber\\
        &\le \frac{1}{|\mathrm{Aut}(G)|^\ell}\left[\left(\gamma \int_{\mathbb{R}^d}\varphi(z)\mathrm{d}z\right)\vee1\right]^{\ell(k-1)+1}k^{k\ell}(\ell!)^ks,
\end{align}
where the last inequality comes from \cite[Proposition~6.1]{schulte2}. 
\end{proof}

\subsection{Hub-to-hub edges}

In the following we study the weighted random connection model such that
$$
\mathbb{P}(x\leftrightarrow y) = \varphi(\kappa(W_x,W_y)/\|x-y\|^d)
$$
for given $x,y\in\eta$, $x\neq y$, with weights $W_x$ and $W_y$. Here, $\kappa: (0,\infty)^2\to(0,\infty)$ is measurable and $\varphi:(0,\infty)\to[0,\infty)$ satisfies
$$
c_\varphi:=\int_{\mathbb{R}^d} \varphi(1/\|v\|^d) \, \mathrm{d}v <\infty.
$$

In the following, we are interested in pairs of edges with large weights that are connected by edges. Let $(t_s)_{s\geq 1}$ be a family of non-negative real numbers. For $s\geq 1$ we study
$$
E_s:= \frac{1}{2} \sum_{(x,y)\in\eta^2_{\neq}} \mathbf{1}\{ x\leftrightarrow y \} \mathbf{1}\bigg\{ \frac{x+y}{2}\in \Gamma_s, W_x,W_y \geq t_s \bigg\},
$$
i.e., we count all edges with vertices having their midpoint in the observation window and weights exceeding $t_s$.

We have that
$$
\mathbb{E}[E_s] = \frac{\gamma^2}{2} \mathbb{E} \int_{(\mathbb{R}^d)^2} \mathbf{1}\bigg\{ \frac{x+y}{2}\in \Gamma_s \bigg\} \mathbf{1}\{W_1,W_2\geq t_s\}  \varphi(\kappa(W_1,W_2)/\|x-y\|^d) \, \mathrm{d}(x,y)
$$
with independent copies $W_1$ and $W_2$ of the typical weight. The substitutions $z=(x+y)/2$ and $u=x-y$ as well as $u'=u/\kappa(W_1,W_2)^{1/d}$ leads to
\begin{align*}
\mathbb{E}[E_s] & = \frac{\gamma^2}{2} \mathbb{E} \int_{(\mathbb{R}^d)^2} \mathbf{1}\{z\in \Gamma_s\} \mathbf{1}\{W_1,W_2\geq t_s\}  \varphi(\kappa(W_1,W_2)/\|u\|^d) \, \mathrm{d}(z,u) \\
& =  \frac{\gamma^2 \operatorname{Vol}(\Gamma_s)}{2} \mathbb{E}\bigg[ \mathbf{1}\{W_1,W_2\geq t_s\}  \int_{\mathbb{R}^d}  \varphi(\kappa(W_1,W_2)/\|u\|^d) \, \mathrm{d}u \bigg] \\
& =  \frac{\gamma^2 s}{2} \mathbb{E}\bigg[ \mathbf{1}\{W_1,W_2\geq t_s\}  \int_{\mathbb{R}^d} \kappa(W_1,W_2) \varphi(1/\|u'\|^d) \, \mathrm{d}u' \bigg] \\
& = \frac{\gamma^2 s}{2} c_\varphi \mathbb{E}[\kappa(W_1,W_2) \mathbf{1}\{W_1,W_2\geq t_s\}].
\end{align*}

\begin{theorem}
Suppose that $\mathbb{E}[\kappa(W_1,W_2)]<\infty$ for independent copies $W_1$ and $W_2$ of $W$.
\begin{itemize}
\item [a)] Assume that $\mathbb{E}[\kappa(W_1,W_2) \kappa(W_1,W_3)]<\infty$ for independent copies $W_1$, $W_2$ and $W_3$ of $W$. For $s\geq 1$ and $Z_s\sim\operatorname{Poisson}(\gamma^2 s c_\varphi \mathbb{E}[\kappa(W_1,W_2) \mathbf{1}\{W_1,W_2\geq t_s\}]/2)$,
$$
\mathrm{d}_{\mathrm{Wass}}(E_s, Z_s) \leq 4 \gamma^3 c_\varphi^2 s \mathbb{E}[ \mathbf{1}\{ W_1,W_2,W_3\geq t_s \} \kappa(W_1,W_2) \kappa(W_1,W_3) ].
$$
\item [b)]
\end{itemize}
\end{theorem}

\begin{proof}
We start with part a). In order to apply Corollary \ref{cor:Poisson_approximation}, we need to compute
\begin{align*}
r_0:= & \gamma^3 \mathbb{E} \int_{(\mathbb{R}^d)^3} \mathbf{1}\bigg\{ \frac{x_1+x_2}{2}, \frac{x_1+x_3}{2}\in \Gamma_s \bigg\} \mathbf{1}\{W_1,W_2,W_3\geq t_s\} \varphi(\kappa(W_1,W_2)/\|x_1-x_2\|^d) \\
& \hspace{3cm} \times \varphi(\kappa(W_1,W_3)/\|x_1-x_3\|^d) \, \mathrm{d}(x_1,x_2,x_3).
\end{align*}
Using substitution, this can be bounded by
\begin{align*}
r_0 & = \gamma^3 \mathbb{E} \int_{(\mathbb{R}^d)^3} \mathbf{1}\bigg\{ \frac{x_1+x_2}{2}, \frac{2x_1+y}{2}\in \Gamma_s \bigg\} \mathbf{1}\{W_1,W_2,W_3\geq t_s\} \varphi(\kappa(W_1,W_2)/\|x_1-x_2\|^d) \\
& \hspace{3cm} \times \varphi(\kappa(W_1,W_3)/\|y\|^d) \, \mathrm{d}(x_1,x_2,y) \\
& \leq \gamma^3 \mathbb{E} \int_{(\mathbb{R}^d)^3} \mathbf{1}\bigg\{ \frac{x_1+x_2}{2}\in \Gamma_s \bigg\} \mathbf{1}\{W_1,W_2,W_3\geq t_s\}\kappa(W_1,W_3) \varphi(\kappa(W_1,W_2)/\|x_1-x_2\|^d) \\
& \hspace{3cm} \times \varphi(1/\|y'\|^d) \, \mathrm{d}(x_1,x_2,y') \\
& = c_\varphi \gamma^3 \mathbb{E} \int_{(\mathbb{R}^d)^2} \mathbf{1}\bigg\{ \frac{x_1+x_2}{2}\in \Gamma_s \bigg\} \mathbf{1}\{W_1,W_2,W_3\geq t_s\}\kappa(W_1,W_3) \varphi(\kappa(W_1,W_2)/\|x_1-x_2\|^d) \, \mathrm{d}(x_1,x_2).
\end{align*}
By the same arguments as for the computation of $\mathbb{E}[E_s]$ above, we obtain
$$
r_0 \leq \gamma^3 c_\varphi^2 s \mathbb{E}[ \mathbf{1}\{ W_1,W_2,W_3\geq t_s \} \kappa(W_1,W_2) \kappa(W_1,W_3)]
$$
Now an application of Corollary \ref{cor:Poisson_approximation} proves part a).

For part b) we evaluate the terms arising in Corollary \ref{normalapp1k2}. By substitution we have that
\begin{align*}
M_{11}(f) & =16\gamma^5 \mathbb{E}\int_{(\mathbb{R}^d)^5} \prod_{i=2}^5 \mathbf{1}\bigg\{ \frac{x_1+x_i}{2}\in\Gamma_s \bigg\} \varphi(\kappa(W_1,W_i)/\|x_1-x_i\|^d) \mathbf{1}\{W_1,W_i\geq t_s\} \, \mathrm{d}(x_1,\hdots,x_5) \\
& =16\gamma^5 \operatorname{Vol}(\Gamma_s) c_\varphi^4 \mathbb{E} \prod_{i=2}^5 \kappa(W_1,W_i) \mathbf{1}\{W_1,W_i\geq t_s\}.
\end{align*}
Similar calculations show that
\begin{align*}
M_{12}(f) & = 16 \gamma^4 s c_\varphi^3 \mathbb{E}[ \kappa(W_1,W_2) \kappa(W_1,W_3) \kappa(W_1,W_4) \mathbf{1}\{W_1,\hdots,W_4\geq t_s\} ] \\
& \quad + 16 \gamma^4 s c_\varphi^3 \mathbb{E}[ \kappa(W_1,W_2) \kappa(W_3,W_4) \kappa(W_1,W_3) \mathbf{1}\{W_1,\hdots,W_4\geq t_s\}] \\
& \quad + 32 \gamma^5 s c_\varphi^4 \mathbb{E}[\kappa(W_1,W_2) \kappa(W_3,W_4) \kappa(W_1,W_5)\kappa(W_3,W_5) \mathbf{1}\{W_1,\hdots,W_5\geq t_s\}] 
\end{align*}
and
\begin{align*}
M_{22}(f) & \leq 48 \gamma^4 s c_\varphi^3 \mathbb{E}[ \kappa(W_1,W_2) \kappa(W_2,W_3) \kappa(W_3,W_4) \kappa(W_4,W_1) \mathbf{1}\{W_1,\hdots,W_4\geq t_s\}] \\
& \quad + 96 \gamma^3 s \|\varphi\|_\infty c_\varphi^2 \mathbb{E}[\kappa(W_1,W_2) \kappa(W_2,W_3) \kappa(W_1,W_3) \mathbf{1}\{W_1,\hdots,W_3\geq t_s\}] \\
& \quad + 48 \gamma^3 s c_\varphi^2  \mathbb{E}[\kappa(W_1,W_2)\kappa(W_1,W_3) \mathbf{1}\{W_1,\hdots,W_3\geq t_s\}] \\
& \quad + 8\gamma s c_\varphi \mathbb{E}[\kappa(W_1,W_2) \mathbf{1}\{W_1,W_2\geq t_s\}]
\end{align*}
\end{proof}

\mathbb{P}(\widetilde{S}^{(n)}\neq\widehat{S}^{(n)}) & \leq \mathbb{P}(\eta_{\mathbb{X}}(A^{(n)}_\ell)\geq 2 \text{ for some } \ell\in\mathbb{N})\nonumber \\
& \leq \sum_{\ell=1}^\infty \mathbb{P}(\eta_{\mathbb{X}}(A^{(n)}_\ell)\geq 2) \leq \sum_{\ell=1}^\infty \lambda(A^{(n)}_\ell)^2 \leq \varepsilon_n^2 \sum_{\ell=1}^\infty \frac{1}{\ell^2}.\label{diff1}
\end{align}
The same argument leads to
\begin{equation}\label{diff2}
\mathbb{P}(\widetilde{\xi}^{(n)}\neq\widehat{\xi}^{(n)}) \leq \varepsilon_n^2 \sum_{\ell=1}^\infty \frac{1}{\ell^2}.
\end{equation}

In the following we denote by $\mathbb{Q}^{m\times m}_{\neq}$ with $m\in\mathbb{N}$ the distribution of $(Y_{i,j})_{i,j\in[k], i\neq j}$ for independent $Y_{i,j}$, $i,j\in[m]$, $i\neq j$, distributed according to $\mathbb{Q}$.

\begin{lemma}\label{lma1}
Let $h: \mathbb{X}^m\times \mathbb{M}_{\neq}^{m\times m}\to \mathbb{R}$ with $m\in\mathbb{N}$ be such that $\|h\|_\infty<\infty$ and
$$
\lambda^m(\{ (x_1,\hdots,x_m)\in\mathbb{X}^m: h(x_1,\hdots,x_k,u)\neq 0 \text{ for some } u\in\mathbb{M}_{\neq}^{m\times m} \})<\infty.
$$
Then,
\begin{align*}
& \lim_{n\to\infty} \int_{\widehat{\mathbb{X}}^m} h(x_1,\hdots,x_m, \tilde{m}^{(n)}(\hat{x}_1,\hdots,\hat{x}_m)) \, \widehat{\lambda}^m(\mathrm{d}(\hat{x}_1,\hdots,\hat{x}_m) ) \\
& = \int_{\mathbb{X}^m} \int_{\mathbb{M}^{m\times m}_{\neq}} h(x_1,\hdots,x_m, u) \, \mathbb{Q}_{\neq}^{m\times m}(\mathrm{d}u) \, \lambda^m(\mathrm{d}(x_1,\hdots,x_m)).
\end{align*}
\end{lemma}

\begin{proof}
Define
$$
\Delta^{(n)}_m := \{ (x_1,\hdots,x_m) \in \mathbb{X}^m: \exists~\ell\in\mathbb{N}, i,j\in[m], i\ne j\text{ with $x_i,x_j\in A^{(n)}_{\ell}$}\}
$$
and
$$
\widehat{\Delta}^{(n)}_m := \{ (\hat{x}_1,\hdots,\hat{x}_m) \in \widehat{\mathbb{X}}^m:  \exists~\ell\in\mathbb{N}, i,j\in[m], i\ne j\text{ with $x_i,x_j\in A^{(n)}_{\ell}$}\}.
$$
For $(\hat{x}_1,\hdots,\hat{x}_m)\in\widehat{\mathbb{X}}^m\backslash\widehat{\Delta}_m^{(n)}$, $\tilde{m}^{(n)}$ assigns different marks to each pair $(i,j)$ with $i,j\in[m]$ and $i\neq j$ so that
\begin{align*}
& \int_{\widehat{\mathbb{X}}^m\backslash\widehat{\Delta}_m^{(n)}} h(x_1,\hdots,x_m, \tilde{m}^{(n)}(\hat{x}_1,\hdots,\hat{x}_m)) \, \widehat{\lambda}^m(\mathrm{d}(\hat{x}_1,\hdots,\hat{x}_m) ) \\
& = \int_{\mathbb{X}^m\backslash\Delta_m^{(n)}} \int_{\mathbb{M}^{m\times m}_{\neq}} h(x_1,\hdots,x_m, u) \, \mathbb{Q}_{\neq}^{m\times m}(\mathrm{d}u) \, \lambda^m(\mathrm{d}(x_1,\hdots,x_m)).
\end{align*}
It follows from \eqref{eqn:monotonicity_partition} and \eqref{eqn:size_partition} that
$$
\mathbf{1}\{ (\hat{x}_1,\hdots,\hat{x}_m)\in\widehat{\mathbb{X}}^m\backslash\widehat{\Delta}_m^{(n)}\} \to 1 \quad \text{as} \quad n\to \infty \quad \text{for} \quad \widehat{\lambda}^m\text{-a.e. } (\hat{x}_1,\hdots,\hat{x}_m)\in\widehat{\mathbb{X}}^m
$$
and
$$
\mathbf{1}\{ (x_1,\hdots,x_m)\in\mathbb{X}^m\backslash\Delta_m^{(n)}\} \to 1 \quad \text{as} \quad n\to \infty \quad \text{for} \quad \lambda^m\text{-a.e. } (x_1,\hdots,x_m)\in\mathbb{X}^m.
$$
Due to the boundedness of $h$ and the assumptions on its support we can apply the dominated convergence theorem and obtain
\begin{align*}
& \lim_{n\to\infty}\int_{\widehat{\mathbb{X}}^m\backslash\widehat{\Delta}_m^{(n)}} h(x_1,\hdots,x_m, \tilde{m}^{(n)}(\hat{x}_1,\hdots,\hat{x}_m)) \, \widehat{\lambda}^m(\mathrm{d}(\hat{x}_1,\hdots,\hat{x}_m) ) \\
& = \lim_{n\to\infty}\int_{\mathbb{X}^m\backslash\Delta_m^{(n)}} \int_{\mathbb{M}^{m\times m}_{\neq}} h(x_1,\hdots,x_m, u) \, \mathbb{Q}_{\neq}^{m\times m}(\mathrm{d}u) \, \lambda^m(\mathrm{d}(x_1,\hdots,x_m)) \\ & = \int_{\mathbb{X}^m} \int_{\mathbb{M}^{m\times m}_{\neq}} h(x_1,\hdots,x_m, u) \, \mathbb{Q}_{\neq}^{m\times m}(\mathrm{d}u) \, \lambda^m(\mathrm{d}(x_1,\hdots,x_m))
\end{align*}
and
$$
\lim_{n\to\infty} \int_{ \widehat{\Delta}^{(n)}_m} h(x_1,\hdots,x_m, \tilde{m}^{(n)}(\hat{x}_1,\hdots,\hat{x}_m)) \, \widehat{\lambda}^m(\mathrm{d}(\hat{x}_1,\hdots,\hat{x}_m) )=0.
$$
Combining the two limits proves the statement.
\end{proof}

\begin{prop}\label{normalU-1}
Let $F\in L^2(\mathbb{P}_\eta)$ be an absolutely convergent $U$-statistics of order $k$ with variance $\Var F>0$ and let $N$ be a standard Gaussian random variable. Then 
\begin{eqnarray}
	\mathrm{d}_{\mathrm{Wass}}\left(\frac{F-\E F}{\sqrt{\Var F}},N\right)&\le&2k^{7/2}\sum_{1\le i\le j\le k}\frac{\sqrt{M_{ij}(f)}}{\Var F},\\

\end{eqnarray}
where 
\begin{equation}
    M_{ij}(f):=\sum_{\pi\in\widetilde{\Pi}_{\ge2}(i,i,j,j)}\int_{X^{|\pi|}}(\overline{f}_i\otimes\overline{f}_i\otimes\overline{f}_j\otimes\overline{f}_j)_{\pi}\mathrm{d}\mu^{|\pi|},
\end{equation}
and 
\begin{equation}
    \overline{f}_i(x_1,\dots,x_i):={k\choose i}\int_{X^{k-i}}|f(x_1,\dots,x_i,y_1,\dots,y_{k-i})|\mu^{k-i}(\mathrm{d}(y_1,\dots,y_{k-i})),
\end{equation}
for $i=1,\dots,k$ and $f_i=0$ for $i>k$.
\end{prop}

\begin{lemma}\label{lem:limits_S_n}
    Let $f: \mathbb{X}^k\times \mathbb{M}^{k\times k}_{\neq}\to\mathbb{R}$ be a measurable function satisfying that $\supp(f)\subseteq B^k\times\mathbb{M}^{k\times k}_{\ne}$ with $\lambda(B)<\infty$ and $\|f\|_{\infty}<\infty$.  
    \begin{enumerate}
        \item[1)] For $p>0$, we have $\E\left[\left|S-\widehat{S}^{(n)}\right|^p\right]\to0$, as $n\to\infty$.
        \item[2)] For $\ell\in\N$, we have 
       \begin{equation} \lim_{n\to\infty}\E\left[(\widehat{S}^{(n)})^{\ell}\right]=\E\left[S^{\ell}\right],
       \end{equation}
       and 
       \begin{eqnarray}
        \E[S^{\ell}]=\sum_{\pi\in\Pi^\ell(k)}\int_{\mathbb{X}^{|\pi|}}\int_{\mathbb{M}^{|\pi|\times|\pi|}_{\ne}}(f^{\otimes\ell})_{\pi}(\bm{x},\bm{u})\mathbb{Q}^{|\pi|\times|\pi|}_{\ne}(\mathrm{d}\bm{u})\lambda^{|\pi|}(\mathrm{d}\bm{x}).
       \end{eqnarray}
    \end{enumerate}
\end{lemma}
\begin{proof}
\begin{enumerate}
        \item[1)] 
According to the coupling $(\widetilde{S}^{(n)},\widehat{S}^{(n)})$ constructed in previous section, together with the Cauchy-Schwarz inequality and \eqref{diff2} it follows that 
\begin{eqnarray*}
    &&\E\left[\left|S-\widehat{S}^{(n)}\right|^p\right]\\
    &=&\E\left[\left|\widetilde{S}^{(n)}-\widehat{S}^{(n)}\right|^p\mathbf{1}_{\{\widetilde{S}^{(n)}\ne\widehat{S}^{(n)}\}}\right]\\
    &=&\E\left[\left|\sum_{(x_1,\hdots,x_k)\in(\eta_{\mathbb{X}})^k_{\neq}} f(x_1,\hdots,x_k, (\widetilde{M}^{(n)}_{x_i,x_j})_{i,j\in[k],i\neq j})-\widehat{S}^{(n)}\right|^p\mathbf{1}_{\{\widetilde{S}^{(n)}\ne\widehat{S}^{(n)}\}}\right]\\
    &=&\E\left[\left|\sum_{(\hat{x}_1,\hdots,\hat{x}_k)\in\widehat{\eta}^k_{\neq}} f(x_1,\hdots,x_k, (\widetilde{M}^{(n)}_{x_i,x_j})_{i,j\in[k],i\neq j})-\widehat{S}^{(n)}\right|^p\mathbf{1}_{\{\widetilde{S}^{(n)}\ne\widehat{S}^{(n)}\}}\right]\\
    &=&\E\left[\left|\sum_{(\hat{x}_1,\hdots,\hat{x}_k)\in\widehat{\eta}^k_{\neq}} \left[f(x_1,\hdots,x_k, (\widetilde{M}^{(n)}_{x_i,x_j})_{i,j\in[k],i\neq j})-f(x_1,\hdots,x_k, \tilde{m}^{(n)}(\hat{x}_1,\hdots,\hat{x}_k))\right]\right|^p\mathbf{1}_{\{\widetilde{S}^{(n)}\ne\widehat{S}^{(n)}\}}\right]\\
    &\le&(2^p\|f\|_{\infty}^p\vee 1)\E\left[\eta(B)^{kp}\mathbf{1}_{\{\widetilde{S}^{(n)}\ne\widehat{S}^{(n)}\}}\right]\\
    &\le&(2^p\|f\|_{\infty}^p\vee 1)\left(\E\eta(B)^{2kp}\right)^{1/2}\left(\IP(\widetilde{S}^{(n)}\ne\widehat{S}^{(n)})\right)^{1/2}\\
    &\le&(2^p\|f\|_{\infty}^p\vee 1)\left(\E\eta(B)^{2kp}\right)^{1/2}\varepsilon_n \sqrt{\sum_{\ell=1}^\infty \frac{1}{\ell^2} }.
\end{eqnarray*}
Therefore, for $p>0$, we have $\lim_{n\to0}\E\left[\left|S-\widehat{S}^{(n)}\right|^p\right]=0$.
\item[2)]
For $\ell\in\N$, $\lim_{n\to0}\E\left[\left|S-\widehat{S}^{(n)}\right|^{\ell}\right]=0$ implies $$\E\left[S^{\ell}\right]=\lim_{n\to\infty}\E\left[(\widehat{S}^{(n)})^{\ell}\right].$$ On the other hand, we have 
\begin{eqnarray*}
\E\left[(\widehat{S}^{(n)})^{\ell}\right]&=&\E\left[\left(\sum_{(\hat{x}_1,\hdots,\hat{x}_k)\in\widehat{\eta}^k_{\neq}} f(x_1,\hdots,x_k, \tilde{m}^{(n)}(\hat{x}_1,\hdots,\hat{x}_k))\right)^{\ell}\right]\\
&=&\E\left[\sum_{\pi\in\Pi^\ell(k)}\sum_{(\hat{x}_1,\dots,\hat{x}_{|\pi|})\in\hat{\eta}^{|\pi|}_{\ne}}(f^{\otimes\ell})_{\pi}(x_1,\dots,x_{|\pi|},\tilde{m}^{(n)}(\hat{x}_1,\hdots,\hat{x}_{|\pi|}))\right]\\
&=&\sum_{\pi\in\Pi^\ell(k)}\left[\int_{\widehat{\mathbb{X}}^{|\pi|}\backslash\widehat{\Delta}_{|\pi|}^{(n)}} (f^{\otimes\ell})_{\pi}(x_1,\dots,x_{|\pi|},\tilde{m}^{(n)}(\hat{x}_1,\hdots,\hat{x}_{|\pi|})) \, \widehat{\lambda}^{|\pi|}(\mathrm{d}(\hat{x}_1,\hdots,\hat{x}_{|\pi|}) )\right.\\
&&\left.\ \ \ \ \ +\int_{\widehat{\Delta}_{|\pi|}^{(n)}}(f^{\otimes\ell})_{\pi}(x_1,\dots,x_{|\pi|},\tilde{m}^{(n)}(\hat{x}_1,\hdots,\hat{x}_{|\pi|})) \, \widehat{\lambda}^{|\pi|}(\mathrm{d}(\hat{x}_1,\hdots,\hat{x}_{|\pi|}) )
\right]\\
&=&\sum_{\pi\in\Pi^\ell(k)}\left[\int_{\mathbb{X}^{|\pi|}\backslash\Delta_{|\pi|}^{(n)}} \int_{\mathbb{M}^{|\pi|\times |\pi|}_{\neq}} (f^{\otimes\ell})_{\pi}(x_1,\dots,x_{|\pi|},u) \, \mathbb{Q}_{\neq}^{|\pi|\times |\pi|}(\mathrm{d}u) \, \lambda^{|\pi|}(\mathrm{d}(x_1,\hdots,x_{|\pi|}))\right.\\
&&\left.\ \ \ \ \ +\int_{\widehat{\Delta}_{|\pi|}^{(n)}}(f^{\otimes\ell})_{\pi}(x_1,\dots,x_{|\pi|},\tilde{m}^{(n)}(\hat{x}_1,\hdots,\hat{x}_{|\pi|})) \, \widehat{\lambda}^{|\pi|}(\mathrm{d}(\hat{x}_1,\hdots,\hat{x}_{|\pi|}) )
\right].
\end{eqnarray*}
Then, by Lemma~\ref{lma1}, we can see that 
\begin{eqnarray*}
    \E\left[S^{\ell}\right]=\sum_{\pi\in\Pi^\ell(k)}\int_{\mathbb{X}^{|\pi|}} \int_{\mathbb{M}^{|\pi|\times |\pi|}_{\neq}} (f^{\otimes\ell})_{\pi}(x_1,\dots,x_{|\pi|},u) \, \mathbb{Q}_{\neq}^{|\pi|\times |\pi|}(\mathrm{d}u) \, \lambda^{|\pi|}(\mathrm{d}(x_1,\hdots,x_{|\pi|})).
\end{eqnarray*}
\end{enumerate}
\end{proof}

\end{enumerate}

\subsection{Proof of Theorem \ref{Poisson_process_approximation}}

\begin{proof}[Proof of Theorem \ref{Poisson_process_approximation}]
The second inequality is an immediate consequence from Lemma \ref{lem:limits_S_n2} and \eqref{eqn:generalised_Mecke_U-statistics}, which yield that
\begin{align*}
& \E\xi(\mathbb{Y})^2 - (\E\xi(\mathbb{Y}))^2 - \E\xi(\mathbb{Y})  \\
& = \Var\xi(\mathbb{Y}) - \E\xi(\mathbb{Y})  \\
& = \sum_{\ell=1}^k \frac{\ell!}{(k!)^2} \binom{k}{\ell}^2 \int_{\mathbb{X}^{2k-\ell}} \int_{\mathbb{M}_{\neq}^{(2k-\ell)\times(2k-\ell)}} \mathbf{1}\{(x_1,\hdots,x_k,(\mathbf{u})_{i,j})_{i,j\in[k],i\neq j})\in\operatorname{dom} g\}  \\
& \hspace{4cm} \times \mathbf{1}\{(x_1,\hdots,x_\ell,x_{k+1},\hdots,x_{2k-\ell},(\mathbf{u})_{i,j})_{i,j\in[\ell]\cup\{k+1,\hdots,2k-\ell\}, i\neq j})\in\operatorname{dom} g\} \\
& \hspace{4cm} \times \mathbb{Q}_{\neq}^{(2k-\ell)\times (2k-\ell)}(d\mathbf{u}) \, \lambda^{2k-\ell}(d(x_1,\hdots,x_{2k-\ell})) \\
& \quad - \frac{1}{k!} \int_{\mathbb{X}^k} \int_{\mathbb{M}_{\neq}^{k\times k}} \mathbf{1}\{(x_1,\hdots,x_k,(\mathbf{u})_{i,j})_{i,j\in[k],i\neq j})\in\operatorname{dom} g\} \, \mathbb{Q}_{\neq}^{k\times k}(d\mathbf{u}) \, \lambda^{k}(d(x_1,\hdots,x_k)) \\
& = \sum_{\ell=1}^{k-1} \frac{\ell!}{(k!)^2} \binom{k}{\ell}^2 \int_{\mathbb{X}^{2k-\ell}} \int_{\mathbb{M}_{\neq}^{(2k-\ell)\times(2k-\ell)}} \mathbf{1}\{(x_1,\hdots,x_k,(\mathbf{u})_{i,j})_{i,j\in[k],i\neq j})\in\operatorname{dom} g\}  \\
& \hspace{4cm} \times \mathbf{1}\{(x_1,\hdots,x_\ell,x_{k+1},\hdots,x_{2k-\ell},(\mathbf{u})_{i,j})_{i,j\in[\ell]\cup\{k+1,\hdots,2k-\ell\}, i\neq j})\in\operatorname{dom} g\} \\
& \hspace{4cm} \times \mathbb{Q}_{\neq}^{(2k-\ell)\times (2k-\ell)}(d\mathbf{u}) \, \lambda^{2k-\ell}(d(x_1,\hdots,x_{2k-\ell})) \\
& \leq \frac{1}{k!} \sum_{\ell=1}^{k-1} \frac{1}{(k-\ell)!} \binom{k}{\ell} r(\operatorname{dom} g) \leq \frac{2^k}{k!} \sum_{\ell=1}^{k-1} r(\operatorname{dom} g).
\end{align*}

Next we show the first inequality for the case that there exists a set $B\subseteq \mathbb{X}$ such that $\operatorname{dom}' g\subseteq B^k$ and $\lambda(B)<\infty$. For $n\in\mathbb{N}$ let $\widetilde{\xi}^{(n)}$ and $\widehat{\xi}^{(n)}$ be as in Section \ref{Poisson_process_approximation} and recall that $\xi\overset{d}{=}\widetilde{\xi}^{(n)}$. The triangle inequality for the Kantorovich-Rubinstein distance yields that
$$
\mathrm{d}_{\rm KR}(\xi,\zeta)=\mathrm{d}_{\rm KR}(\widetilde{\xi}^{(n)},\zeta) \leq \mathrm{d}_{\rm KR}(\widetilde{\xi}^{(n)},\widehat{\xi}^{(n)}) + \mathrm{d}_{\rm KR}(\widehat{\xi}^{(n)},\zeta).
$$
From the definition of the Kantorovich-Rubinstein distance, the Cauchy-Schwarz inequality and \eqref{diff2} it follows that
\begin{align*}
	\mathrm{d}_{\rm KR}(\widetilde{\xi}^{(n)},\widehat{\xi}^{(n)}) & \leq \E\big[ \mathbf{1}\{ \widetilde{\xi}^{(n)}\neq\widehat{\xi}^{(n)} \} (\widetilde{\xi}^{(n)}(\mathbb{Y})+\widehat{\xi}^{(n)}(\mathbb{Y})) \big] \\
& \leq \mathbb{P}( \widetilde{\xi}^{(n)}\neq\widehat{\xi}^{(n)} )^{1/2} \big( (\E \widetilde{\xi}^{(n)}(\mathbb{Y})^2)^{1/2} + (\E \widetilde{\xi}^{(n)}(\mathbb{Y})^2)^{1/2} \big) \\
& \leq \varepsilon_n \sqrt{\sum_{\ell=1}^\infty \frac{1}{\ell^2} } \big( (\E \widetilde{\xi}^{(n)}(\mathbb{Y})^2)^{1/2} + (\E \widetilde{\xi}^{(n)}(\mathbb{Y})^2)^{1/2} \big).
\end{align*}
Because of $\E \widetilde{\xi}^{(n)}(\mathbb{Y})^2, \E \widetilde{\xi}^{(n)}(\mathbb{Y})^2 \leq \E \eta(B)^k<\infty$ by the assumption on $g$, we see that
$$
\lim_{n\to\infty} \mathrm{d}_{\rm KR}(\widetilde{\xi}^{(n)},\widehat{\xi}^{(n)})=0.
$$
From \cite[Theorem 3.1]{dst} we obtain that
$$
\mathrm{d}_{\rm KR}(\widehat{\xi}^{(n)},\zeta) \leq \mathrm{d}_{\rm TV}(\widehat{\mathbf{L}}^{(n)},\mathbf{M}) + 2 (\E \widehat{\xi}^{(n)}(\mathbb{Y})^2 - (\E \widehat{\xi}^{(n)}(\mathbb{Y}))^2 - \E \widehat{\xi}^{(n)}(\mathbb{Y}) ),
$$
where $\widehat{\mathbf{L}}^{(n)}$ is the intensity measure of $\widehat{\xi}^{(n)}$. Note that
$$
\widehat{\mathbf{L}}^{(n)} = \mathbf{L}.
$$
Combining this with Lemma \ref{lem:limits_S_n}, which is applicable as $\widehat{\xi}^{(n)}(\mathbb{Y})$ is a $U$-statistic, yields that
$$
\limsup_{n\to\infty} \mathrm{d}_{\rm KR}(\widehat{\xi}^{(n)},\zeta) \leq \mathrm{d}_{\rm TV}(\mathbf{L},\mathbf{M}) + 2 (\E \widehat{\xi}(\mathbb{Y})^2 - (\E \widehat{\xi}(\mathbb{Y}))^2 - \E \widehat{\xi}(\mathbb{Y}) ),
$$
which is the desired statement under the additional assumption on $\operatorname{dom}'g$.

Next we consider the situation of general $g$. Since $\lambda$ is $\sigma$-finite, there exists an increasing sequence $(B_m)_{m\in\mathbb{N}}$ such that $\lambda(B_m)<\infty$ for all $m\in\mathbb{N}$ and $\bigcup_{m=1}^\infty B_m=\mathbb{X}$. For $m\in\mathbb{N}$ we define
$$
\xi_m=\frac{1}{k!}\sum_{(x_1,\hdots,x_k)\in\eta^k_{\neq}\cap \operatorname{dom}'g \cap B_m^k} \delta_{f(x_1,\hdots,x_k,(\mathcal{M}_{i,j})_{i,j\in[k],i\neq j})}
$$
and denote by $\mathbf{L}_m$ the intensity measure of $\xi_m$. From the triangle inequality and the first part of the proof we know that
\begin{align*}
\mathrm{d}_{\rm KR}(\xi,\zeta) & \leq \mathrm{d}_{\rm KR}(\xi_m,\xi) + \mathrm{d}_{\rm KR}(\xi_m,\zeta) \\
			       & \leq  \mathrm{d}_{\rm KR}(\xi_m,\xi) + \mathrm{d}_{\rm TV}(\mathbf{L}_m,\mathbf{M}) + 2(\E\xi_m(\mathbb{Y})^2 - (\E\xi_m(\mathbb{Y}))^2 - \E\xi_m(\mathbb{Y}) )
\end{align*}
Because every point of $\xi_m$ is also contained in $\xi$, we deduce that
$$
\mathrm{d}_{\rm KR}(\xi_m,\xi) \leq \E[|\xi(\mathbb{Y})-\xi_m(\mathbb{Y})|] = \E[\xi(\mathbb{Y})] - \E[\xi_m(\mathbb{Y})].
$$
and
$$
\mathrm{d}_{\rm TV}(\mathbf{L}_m,\mathbf{L}) \leq \E[\xi(\mathbb{Y})] - \E[\xi_m(\mathbb{Y})].
$$
Together with the triangle inequality, the latter leads to
$$
\mathrm{d}_{\rm TV}(\mathbf{L}_m,\mathbf{M}) \leq \mathrm{d}_{\rm TV}(\mathbf{L}_m,\mathbf{L}) + \mathrm{d}_{\rm TV}(\mathbf{L},\mathbf{M}) \leq \mathrm{d}_{\rm TV}(\mathbf{L},\mathbf{M})  +  \E[\xi(\mathbb{Y})] - \E[\xi_m(\mathbb{Y})] ,
$$
so that
$$
\mathrm{d}_{\rm KR}(\xi,\zeta) \leq \mathrm{d}_{\rm TV}(\mathbf{L},\mathbf{M})  + 2 (\E[\xi(\mathbb{Y})] - \E[\xi_m(\mathbb{Y})]) + 2(\E\xi_m(\mathbb{Y})^2 - (\E\xi_m(\mathbb{Y}))^2 - \E\xi_m(\mathbb{Y}) ).
$$
Since, by the dominated convergence theorem, $\E\xi_m(\mathbb{Y})^2\to \E\xi(\mathbb{Y})^2$ and $\E\xi_m(\mathbb{Y})\to \E\xi(\mathbb{Y})$ as $m\to\infty$, we obtain
$$
\mathrm{d}_{\rm KR}(\xi,\zeta) \leq \mathrm{d}_{\rm TV}(\mathbf{L},\mathbf{M}) + 2(\E\xi(\mathbb{Y})^2 - (\E\xi(\mathbb{Y}))^2 - \E\xi(\mathbb{Y}) ),
$$
which completes the proof.
\end{proof}

\subsection{Expectation and variance asymptotics}

Consider the statistics

$$L_\rho (\tau):=\frac{1}{2} \sum_{(z_1, z_2) \in \eta^2_{\neq} \cap B(0, \rho)^2} \mathbf{1}\{ z_1 \leftrightarrow z_2 \} \| z_1- z_2 \|^\tau, \quad \tau \in \mathbb{R}.$$

Proceeding as in \cite[Theorem 3.1]{reitznerschultethaele} gives the following formula for its expectation structure. 

\begin{lemma} \label{lem:expectedlength}
    Assume that $\tau > -d$. It holds
    $$ \mathbb{E}[L_\rho (\tau)]= \frac{\gamma^2}{2} \int_0^\infty \bar{\varphi} (r) r^{\tau + d -1} \int_{\mathbb{S}^{d-1}
} f_\rho (r u) \mathrm{d}u \mathrm{d}r,$$
where
$$f_\rho (ru) := \lambda_d \left( B(0,\rho) \cap (B(0,\rho) + ru) \right)$$
and $\mathrm{d}u$ stands for the infinitesimal element of the spherical Lebesgue measure. Moreover, it holds
\begin{align*}
    & \frac{\gamma^2}{2} d \kappa_d \lambda_d(B(0,\rho)) \int_0^{2\rho} \bar{\varphi} (r) r^{\tau + d -1} \mathrm{d}r -\gamma^2\kappa_{d-1} \operatorname{surf}(B(0,\rho)) \int_0^{2\rho} \bar{\varphi} (r) r^{\tau + d } \mathrm{d}r\\
    & \leq \mathbb{E}[L_\rho (\tau)] \leq \frac{\gamma^2}{2} d \kappa_d \lambda_d(B(0,\rho)) \int_0^{2\rho} \bar{\varphi} (r) r^{\tau + d -1} \mathrm{d}r,
\end{align*}
where $\kappa_d$ denotes the volume of the $d$-dimensional unit ball and $\operatorname{surf}(B(0,\rho))$ denotes the surface area of the ball  $B(0,\rho)$.
\end{lemma}

Note that Lemma \ref{lem:expectedlength} shows that the asymptotics of the expectation of power-weighted edge-lengths depends on whether the integral
$$\int_0^\infty \bar{\varphi} (r) r^{\tau + d -1} \mathrm{d} r $$
is finite or not. If we assume that $\bar{\varphi}$ has polynomial decay $\bar{\varphi}(r) \sim r^{-\alpha}$, $r \to \infty$, for some exponent $\alpha$, this translates to whether $\alpha>\tau +d$ or $\alpha\leq\tau +d$.

with
\begin{align*}
   d \kappa_d \lambda_d(B(0,\rho))  -\kappa_{d-1} \operatorname{surf}(B(0,\rho)) r\leq \int_{\mathbb{S}^{d-1}} 
       \mathrm{Vol}\!\big(B(0,\rho)\cap (B(0,\rho)+ru)\big)\,
        \mathrm{d}u \leq d \kappa_d \lambda_d(B(0,\rho)) 
\end{align*}
so that the variance asymptotics depends on whether
$$\int_0^\infty \bar{\varphi} (r) r^{2\tau + d -1} \mathrm{d} r< \infty$$ 
or not. Moreover the asymptotics of $I(\rho)$, as $\rho \to \infty$, is
$$ C_1 \rho^d (C_1 + \rho^{2\tau + d -\alpha} )\leq I(\rho) \leq C_2 \rho^d (C_2 + \rho^{2\tau + d -\alpha} ) $$
for some constants $C_1, C_2>0$, if $\alpha \neq 2\tau +d$, and
$$ C_1 \rho^d (C_1 + \log \rho )\leq I(\rho) \leq C_2 \rho^d (C_2 + \log \rho ) $$
if $\alpha = 2\tau +d$.

Similarly, considering the second term on the right-hand side of \eqref{varformula} we obtain
\begin{align*}
J(\rho) 
  &:= \int_{B(0,\rho)}\!
     \Bigg(\int_{B(0,\rho)} \varphi(x_1-y)\,\|y-x_1\|^{\tau}\,\mathrm{d}x_1\Bigg)
     \Bigg(\int_{B(0,\rho)} \varphi(x_2-y)\,\|y-x_2\|^{\tau}\,\mathrm{d}x_2\Bigg)
     \mathrm{d}y \\[0.4em]
  &= \int_{B(0,\rho)}\!
     \int_{\mathbb{R}^d}\!\int_{\mathbb{R}^d}
        \varphi(z_1)g(z_2)\,\|z_1\|^{\tau}\|z_2\|^{\tau}\,
        \mathbf{1}_{\{y+z_1\in B(0,\rho)\}}\,
        \mathbf{1}_{\{y+z_2\in B(0,\rho)\}}\,
        \mathrm{d}z_1\,\mathrm{d}z_2\,\mathrm{d}y \\[0.4em]
  &= \int_{\mathbb{R}^d}\!\int_{\mathbb{R}^d}
        \varphi(z_1)\varphi(z_2)\,\|z_1\|^{\tau}\|z_2\|^{\tau}\,
        \mathrm{Vol}\!\big(B(0,\rho)\cap(B(0,\rho)+z_1)\cap(B(0,\rho)+z_2)\big)\,
        \mathrm{d}z_1\,\mathrm{d}z_2 \\[0.6em]
  &= \int_{0}^{2\rho}\!\!\int_{0}^{2\rho}
     \int_{\mathbb{S}^{d-1}}\!\int_{\mathbb{S}^{d-1}}
        \varphi(r_1)\varphi(r_2)\,r_1^{\tau}r_2^{\tau}\,
      \mathrm{Vol}\!\big(B(0,\rho)\cap(B(0,\rho)+r u_1)\cap(B(0,\rho)+r u_2)\big)\,
        \mathrm{d}u_1\,\mathrm{d}u_2\,
        \mathrm{d}r_1\,\mathrm{d}r_2 
\end{align*}
with the asymptotic bound
$$ \frac{1}{C} \rho^d\leq \mathrm{Vol}\!\big(B(0,\rho)\cap(B(0,\rho)+r u_1)\cap(B(0,\rho)+r u_2)\big) \leq {C} \rho^d ,$$
so that the assertion follows easily.
    
\end{proof}

From the statements in Lemma \ref{lem:varlength} we observe that the variance asymptotics is not finite.

\subsection{Regime with central limit theorem}

The next theorem illustrates that there is a central limit theorem in the regime with polynomial decay for exponents $\alpha$ satisfying $\alpha> 2\tau + d$.

\begin{theorem} \label{CLT:sumsedgelengths}
    Let $\tau > -d$. Assume $\bar{\varphi}$ has polynomial decay $\bar{\varphi}(r) \sim r^{-\alpha}$, $r \to \infty$, with (positive) exponent $\alpha \in (2\tau + d, \infty)$. Then there is a constant $C \in (0, \infty)$ independent of $\rho$ such that
    \begin{align*}
	\mathrm{d}_{\mathrm{Wass}}\left(\frac{L_\rho(\tau)-\E L_\rho(\tau)}{\sqrt{\Var L_\rho(\tau)}},N\right) & \le C \max\{ \rho^{-\frac{d}{2}} , \rho^{2\tau -\frac{\alpha}{2}} \},\\
	\mathrm{d}_{\mathrm{K}}\left(\frac{L_\rho(\tau)-\E L_\rho(\tau)}{\sqrt{\Var L_\rho(\tau)}},N\right) & \le  C \max\{ \rho^{-\frac{d}{2}} , \rho^{2\tau -\frac{\alpha}{2}} \} .
\end{align*}
\end{theorem}

\begin{proof}
    The theorem will be a consequence of Corollary \ref{normalapp1k2}. In this proof we denote by $c$ an unspecified positive constant that might change from line to line. We can estimate each expression $M_{ij}(f)$ in the latter corollary as follows. Considering the term $M_{11}(f)$ we have
    \begin{align*}
M_{11}(f)
&= c \int_{B(0,\rho)^5} \prod_{j=2}^{5} \|x_1-x_j\|^{\tau}\,\varphi(x_1-x_j)\,\mathrm{d}x_1\cdots \mathrm{d}x_5 \\[4pt]
&= c\int_{B(0,\rho)} \left( \int_{B(0,\rho)^4} \prod_{j=2}^{5}  \|x_1-x_j \|^{\tau}\,\varphi(x_1-x_j)\,\mathrm{d}x_2\cdots \mathrm{d}x_5 \right) \mathrm{d}x_1 \\[4pt]
&= \int_{B(0,\rho)} \left( \int_{B(-x_1,\rho)} |u|^{\tau}\,\varphi(u)\,\mathrm{d}u \right)^{4} \mathrm{d}x_1,
\end{align*}
where $B(-x_1,\rho) \;:=\; \{\, u \in \mathbb{R}^d : |u + x_1| \le \rho \,\}$ and we used the substitution $u=x_1-x_j$ for each $j\ge 2$. S\text{ince }$B(-x_1,\rho)\subseteq B(0,2\rho)$ \text{ for }$|x_1|\le\rho$, the latter can be further estimated by
\begin{align*}
&\int_{B(0,\rho)} \left( \int_{B(0,2\rho)} |u|^{\tau}\,\varphi(u)\,\mathrm{d}u \right)^{4} \mathrm{d}x_1= \mathrm{Vol}\big(B(0,\rho)\big)\,\left( \int_{B(0,2\rho)} |u|^{\tau}\,\varphi(u)\,\mathrm{d}u \right)^{4}
\end{align*}
\text{Assuming }$\alpha>2\tau+d$ we have
\begin{align*}
\int_{\mathbb{R}^d} |u|^{\tau}\varphi(u)\,\mathrm{d}u =:\kappa<\infty.
\end{align*}
Consequently
$$\int_{B(0,2\rho)} |u|^{\tau}\varphi(u)\,\mathrm{d}u \uparrow \kappa,$$
and therefore
\begin{align} \label{cltedgelengthsm11}

\;=\; C\,\rho^{d}\qquad(\text{for some }C>0).
\end{align}

\subsubsection{Truncation}

For $a>0$ define
$$
L_{\rho,a} (\tau):=\frac{1}{2} \sum_{(z_1, z_2) \in \eta^2_{\neq} \cap B(0, \rho)^2} \mathbf{1}\{ z_1 \leftrightarrow z_2, \|z_1-z_2\|\leq a \} \| z_1- z_2 \|^\tau.
$$
We aim to estimate
$$
\frac{\mathbb{E}[(L_{\rho,a} (\tau)-L_{\rho} (\tau))^2]}{\rho^d}. %\leq c a^{-...}
$$
By definition we have
\begin{align*}
L_{\rho,a}(\tau)-L_{\rho}(\tau)
&= \frac{1}{2}\sum_{(z_1,z_2)\in\eta^2_{\neq}\cap B(0,\rho)^2}
\mathbf{1}\{z_1\leftrightarrow z_2\}
\|z_1-z_2\|^{\tau}
\big(\mathbf{1}_{\{\|z_1-z_2\|\le a\}}-1\big) \\[0.4em]
&= -\frac{1}{2}\sum_{(z_1,z_2)\in\eta^2_{\neq}\cap B(0,\rho)^2}
\mathbf{1}\{z_1\leftrightarrow z_2,\ \|z_1-z_2\|>a\}\,
\|z_1-z_2\|^{\tau},
\end{align*}
and
\begin{align*}
& (L_{\rho,a}(\tau)-L_{\rho}(\tau))^2\\
& = \frac{1}{4}\!\!\sum_{(z_1,z_2),(z_3,z_4)\in\eta^2_{\neq}\cap B(0,\rho)^4}
\!\!\!\!\mathbf{1}\{z_1\leftrightarrow z_2,\|z_1-z_2\|>a\}
\mathbf{1}\{z_3\leftrightarrow z_4,\|z_3-z_4\|>a\}
\|z_1-z_2\|^{\tau}\|z_3-z_4\|^{\tau},
\end{align*}
so that
\begin{align*}
\mathbb{E}\!\left[(L_{\rho,a}(\tau)-L_{\rho}(\tau))^2\right]
&= C_1 I_a(\rho) + C_2 J_a(\rho),
\end{align*}
where, by Mecke’s formula,
\begin{align*}
I_a(\rho)
&= \int_{B(0,\rho)}\!\!\int_{B(0,\rho)}
\mathbf{1}_{\{\|x-y\|>a\}}\,
\varphi(x-y)\,\|x-y\|^{2\tau}\,\mathrm{d}x\,\mathrm{d}y, \\[0.3em]
J_a(\rho)
&= \int_{B(0,\rho)}\!
\Bigg(\int_{B(0,\rho)}
\mathbf{1}_{\{\|x_1-y\|>a\}}\,
\varphi(x_1-y)\,\|x_1-y\|^{\tau}\,\mathrm{d}x_1\Bigg)^{\!2}\mathrm{d}y.
\end{align*}
Assuming $\varphi(r)\sim C r^{-\alpha}$ with $\alpha>2\tau+d$, the tails are integrable and the same calculations as before show that
\begin{align*}
I_a(\rho),\,J_a(\rho)\;\leq \;c\rho^{d}\,a^{2\tau+d-\alpha},
\end{align*}
so that we have proven the following.

\begin{lemma}\label{truncationlemma1}
    There exists a constant $c \in (0,\infty)$ not depending on $a$ such that
    $$\frac{\mathbb{E}[(L_{\rho,a} (\tau)-L_{\rho} (\tau))^2]}{\rho^d} \leq c a^{2\tau+d-\alpha} .$$
\end{lemma}

Next we prove that $L_{\rho,a} (\tau)$ satisfies a CLT for every $a>0$ and the regime $2\tau+d<\alpha$. The strategy is identical to the proof of Theorem \ref{CLT:sumsedgelengths}, with the key difference that the truncation will give us better (quantitative convergence) bounds.

\begin{lemma}\label{lem:CLT_truncated}
Let $\tau > -d$ and $a>0$. 
Assume that $\bar{\varphi}$ has polynomial decay $\bar{\varphi}(r)\sim r^{-\alpha}$ as $r\to\infty$ with exponent $\alpha>2\tau+d$. 
Then there exists a constant $C\in(0,\infty)$, independent of $\rho$, such that
\begin{align*}
\mathrm{d}_{\mathrm{Wass}}\!\left(
\frac{L_{\rho,a}(\tau)-\E L_{\rho,a}(\tau)}{\sqrt{\Var L_{\rho,a}(\tau)}},
N
\right)
&\le C\,\rho^{-d/2},\\[0.4em]
\mathrm{d}_{\mathrm{K}}\!\left(
\frac{L_{\rho,a}(\tau)-\E L_{\rho,a}(\tau)}{\sqrt{\Var L_{\rho,a}(\tau)}},
N
\right)
&\le C\,\rho^{-d/2}.
\end{align*}
\end{lemma}

\begin{proof}
The proof follows the same steps as in Theorem~\ref{CLT:sumsedgelengths}, 
with the additional indicator $\mathbf{1}_{\{\|x_1-x_2\|\le a\}}$ restricting all integrals to a bounded region. 
For instance,
\[
M_{11}(f_a)
= c \int_{B(0,\rho)}
\!\left(
\int_{B(-x_1,\rho)\cap B(0,a)} |u|^{\tau}\varphi(u)\,\mathrm{d}u
\right)^{\!4}
\mathrm{d}x_1
\le C\,\rho^d,
\]
since $\int_{B(0,a)}|u|^{\tau}\varphi(u)\,\mathrm{d}u<\infty$.
Analogous bounds hold for $M_{12}(f_a)$ and $M_{22}(f_a)$.
Thus, all moment terms in Corollary~\ref{normalapp1k2} are of order $\rho^d$, 
and by Lemma~\ref{lem:varlength} the variance satisfies $\Var[L_{\rho,a}(\tau)]\geq c\rho^d$.
Applying Corollary~\ref{normalapp1k2} yields the stated bounds with rate $\rho^{-d/2}$.
\end{proof}

Thus, Lemma \ref{truncationlemma1} and Lemma \ref{lem:CLT_truncated} yield the following theorem, giving a CLT in the entire regime $\alpha>2\tau+d$ for $L_{\rho}(\tau)$.

\begin{theorem}\label{thm:CLTedgelengths_full}
Let $\tau > -d$ and assume that $\bar{\varphi}(r)\sim r^{-\alpha}$ as $r\to\infty$ with exponent $\alpha>2\tau+d$. 
Then, as $\rho\to\infty$,
\[
\frac{L_{\rho}(\tau)-\E[L_{\rho}(\tau)]}{\sqrt{\Var[L_{\rho}(\tau)]}}
\;\xrightarrow{d}\;
N.
\]
\end{theorem}
