\section{$U$-statistics of edge-marked Poisson processes}



\subsection{Edge-marked Poisson processes}

Although we are mostly interested in the Random Connection Model, we derive general results for certain functionals and point processes constructed from so-called edge-marked Poisson processes. The idea of an edge-marked Poisson process is to take a Poisson process and to assign an independent and identically distributed mark to every pair of points. The random connection model can be seen as such an edge-marked Poisson process. We take the Poisson process of the vertices and assign to each pair of vertices $x$ and $y$ an independent random variable $U_{x,y}$ that is uniformly distributed on $[0,1]$. Then we connect $x$ and $y$ by an edge if and only if $U_{x,y}\leq \varphi(x,y)$, where $\varphi$ is the connection function.

In the following, we introduce edge-marked Poisson processes on a general measurable space. To this end let $(\mathbb{X},\mathcal{X})$ be a measurable space and let $\lambda$ be a $\sigma$-finite non-atomic measure on $\mathbb{X}$. For $\mathbb{X}$ and $\lambda$ we always assume that for each $n\in\mathbb{N}$ there exists a partition $(A_\ell^{(n)})_{\ell\in\mathbb{N}}$ of $\mathbb{X}$, i.e., $A^{(n)}_{\ell_1}\cap A^{(n)}_{\ell_2}=\emptyset$ for distinct $\ell_1,\ell_2\in\mathbb{N}$ and $\bigcup_{\ell=1}^\infty A^{(n)}_\ell=\mathbb{X}$, such that the following conditions are satisfied:
\begin{itemize}
\item For $x\in\mathbb{X}$ and $n\in\mathbb{N}$ let $A^{(n)}(x)$ be the unique $A^{(n)}_\ell$ such that $x\in A^{(n)}_\ell$. Then,
\begin{equation}\label{eqn:monotonicity_partition}
A^{(n)}(x)\subseteq A^{(m)}(x)
\end{equation}
for all $x\in\mathbb{X}$ and $m,n\in\mathbb{N}$ with $m\leq n$.
\item There exists a sequence $(\varepsilon_n)_{n\in\mathbb{N}}$ such that $\lim_{n\to\infty}\varepsilon_n=0$ and
\begin{equation}\label{eqn:size_partition}
\lambda(A_\ell^{(n)}) \leq \frac{\varepsilon_n}{\ell}
\end{equation}
for all $\ell,n\in\mathbb{N}$.
\item There exists a measurable function $\operatorname{dist}:\mathbb{X}\times \mathbb{X}\to[0,\infty)$ such that 
\begin{equation}\label{eqn:uniqueness_distances}
\lambda^3\left(\left\{(x,y,z)\in \mathbb{X}\times A^{(n)}_\ell\times A_\ell^{(n)}: \operatorname{dist}(x,y)=\operatorname{dist}(x,z)\right\}\right)=0
\end{equation}
for all $\ell,n\in\mathbb{N}$.
\end{itemize}

Let us briefly discuss choices of $\mathbb{X}$ and $\lambda$ that are relevant for studying the random connection model. For the stationary random connection model without weights one chooses $\mathbb{X}=\mathbb{R}^d$ and $\lambda=\gamma\lambda_d$, where $\lambda_d$ is the $d$-dimensional Lebesgue measure and $\gamma>0$ is an intensity parameter. For the case of the random connection model with weights we let $\mathbb{X}=\mathbb{R}^d\times [0,\infty)$ and $\lambda=\gamma\lambda_d\otimes \mathbb{W}$ with $\gamma$ and $\lambda_d$ as before and the weight distribution $\mathbb{W}$. In case one is only interested in the random connection model in some observation window $K$ one replaces $\mathbb{R}^d$ by $K$ and the Lebesgue measure by the restriction of the Lebesgue measure.  

Let $(\mathbb{X},\mathcal{X})$ be a measurable space and let $\eta$ be a Poisson process on $\mathbb{X}$ with the $\sigma$-finite non-atomic intensity measure $\lambda$. We assume that $\eta$ can be almost surely written as 
$$
\eta=\sum_{i\in I} \delta_{X_i}
$$
with random elements $(X_i)_{i\in I}$ on $\mathbb{X}$ and a random index set $I\subseteq\mathbb{N}$, where $\delta_x$ denotes the Dirac measure concentrated on $x$. 

Moreover, let $(\mathbb{M},\mathcal{M})$ be a measurable space with a probability measure $\mathbb{Q}$. Let $(M_{i,j})_{i,j\in\mathbb{N},i\neq j}$ be a family of independent random elements on $\mathbb{M}$ distributed according to $\mathbb{Q}$. Now we assign the mark $M_{i,j}$ to the pair $(X_i,X_j)$ for $i,j\in I$ with $i\neq j$. For $y,z\in\eta$ with $y\neq z$ we denote by $M_{y,z}$ the mark of the pair $(y,z)$. Note that the pairs $(y,z)$ and $(z,y)$ have different marks.



\subsection{$U$-statistics and moment formulae}
For $k\in\mathbb{N}$ we use the short hand notation $[k]:=\{1,\hdots,k\}$ and define
$$
\mathbb{M}^{k\times k}_{\neq}:=\{(u_{i,j})_{i,j\in[k],i\neq j}: u_{i,j}\in \mathbb{M} \text{ for } i,j\in[k],i\neq j\}.
$$
Moreover let $\eta^k_{\neq}$ be the set of all $k$-tuples of distinct points of $\eta$.



For $f: \mathbb{X}^k\times \mathbb{M}^{k\times k}_{\neq}\to\mathbb{R}$ with $k\in\mathbb{N}$ we are interested in random variables of the following form:
\begin{equation}\label{def:ustat}
S=\sum_{(x_1,\hdots,x_k)\in\eta^k_{\neq}} f(x_1,\hdots,x_k, (M_{x_i,x_j})_{i,j\in[k], i\neq j}).
\end{equation}
We also denote 
\begin{equation}\label{def-integ1}
    \overline{S}:=\sum_{(x_1,\hdots,x_k)\in\eta^k_{\neq}} \left|f(x_1,\hdots,x_k, (M_{x_i,x_j})_{i,j\in[k], i\neq j})\right|.
\end{equation}
\begin{prop}
    % Let $f: \mathbb{X}^k\times \mathbb{M}^{k\times k}_{\neq}\to\mathbb{R}$ with $k\in\mathbb{N}$ be a measurable function. 
    \begin{enumerate}
    \item[]
    \item[a)]
    Suppose $\E(\overline{S})<\infty$, then we have 
\begin{equation}\label{eqn:generalised_Mecke_U-statistics}
\E S = \int_{\mathbb{X}^k}\int_{\mathbb{M}^{k\times k}_{\neq}} f(x_1,\hdots,x_k,\mathbf{u}) \, \mathbb{Q}_{\neq}^{k\times k}(d\mathbf{u}) \, \lambda^k(d(x_1,\hdots,x_k)).
\end{equation}
\item[b)] Suppose $\E\left(\overline{S}^2\right)<\infty$, we have 
\begin{align}\label{eqn:variance}
\Var S =\sum_{\ell=1}^k{k\choose \ell}^2\ell! &\int_{\mathbb{X}^{2k-\ell}}\int_{\mathbb{M}^{(2k-\ell)\times (2k-\ell)}_{\neq}} f(x_1,\hdots,x_k,(u_{i,j})_{i,j\in[k],i\ne j}) \nonumber\\
&f(x_1,\dots,x_{\ell},x_{k+1},\dots,x_{2k-\ell},(u_{i,j})_{i,j\in[\ell]\cup\{k+1,\dots,2k-\ell\},i\ne j}) )\\
&\mathbb{Q}_{\neq}^{(2k-\ell)\times (2k-\ell)}(d\mathbf{u}) \, \lambda^{2k-\ell}(d(x_1,\hdots,x_{2k-\ell})).\nonumber
\end{align}
\end{enumerate}
\end{prop}
For higher moments, we need a few more notation about partitions. Let $A$ be a non-empty set. A partition $\sigma$ of $A$ is a collection $\{B_1,\dots,B_s\}$ of pairwise disjoint non-empty subsets $B_i\subseteq A$, called blocks, such that $B_1\cup\cdots\cup B_s=A$. The number $s$ of blocks in $\sigma$ is denoted by $|\sigma|$.
% For any $n\in\N$, we denote $[n]:=\{1,\dots,n\}$.
Let $t\in\N$ and $q\in\N$. Set $J_\ell:=\{(\ell-1)q+1,\dots,\ell q\}$, for $\ell=1,\dots,t.$ By $\Pi^t(q)$ we denote the set of all partitions $\sigma$ such that $|B\cap J_\ell|\le1$ for all $\ell\in[t]$ and $B\in\sigma$.

Every partition $\sigma\in\Pi^t(q)$ induces a partition $\sigma^*$ of $[t]$ in the following way: $i,j\in[t]$ are in the same block of $\sigma^*$ whenever there exists a block $B\in\sigma$ such that $|B\cap J_i|=|B\cap J_j|=1$. Let $\widetilde{\Pi}^t(q)$ be the collection of all partitions $\sigma\in\Pi^t(q)$ such that $|\sigma^*|=1$. Note that partitions in $\Pi^t(q)$ and $\widetilde{\Pi}^t(q)$ are known as non-flat and connected non-flat respectively, see \cite[Chapter~4]{peccati2011}. 
We denote by $\overline{\Pi}^t(q)$ the collection of all partitions $\sigma\in\Pi^t(q)$ such that for each $\ell\in[t]$ there exists a block $B\in\sigma$ with $|B|\ge2$ and $B\cap J_\ell\ne\emptyset$.
%By $\Pi_{\ge2}(q_1,\dots,q_t)$ and $\widetilde{\Pi}_{\ge2}(q_1,\dots,q_t)$, we denote the sets of all $\sigma\in\Pi(q_1,\dots,q_t)$ and $\sigma\in\widetilde{\Pi}(q_1,\dots,q_t)$ such that $|B|\ge2$ for every $B\in\sigma$. In the case $q_1=\dots=q_t=q$, we write $\Pi^t(q)$, $\widetilde{\Pi}^t(q)$, $\Pi^t_{\ge2}(q)$, $\widetilde{\Pi}^t_{\ge2}(q)$ and $\overline{\Pi}^t(q)$ instead of $\Pi(q_1,\dots,q_t)$, $\widetilde{\Pi}(q_1,\dots,q_t)$, $\Pi_{\ge2}(q_1,\dots,q_t)$, $\widetilde{\Pi}_{\ge2}(q_1,\dots,q_t)$ and $\overline{\Pi}(q_1,\dots,q_t)$ respectively.

Let $(S,\mathcal{S})$ be an arbitrary measurable space. For functions $f^{(\ell)}:S^{q}\to\R$, $\ell\in[t]$, we define their tensor product $\otimes_{\ell=1}^tf^{(\ell)}:S^{tq}\to\R$ by 
$$\otimes_{\ell=1}^tf^{(\ell)}(x_1,\dots,x_{tq}):=\prod_{\ell=1}^tf^{(\ell)}(x_{(\ell-1)q+1},\dots,x_{\ell q}).$$
For each $\sigma\in\Pi^t(q)$, the function $(\otimes_{\ell=1}^tf^{(\ell)})_\sigma:S^{|\sigma|}\to\mathbb{R}$ is obtained by replacing all variables that belong to the same block in $\sigma$ by a new common variable.
\begin{prop}\label{prop:moments}
    Suppose that $\E\left(\overline{S}^{\ell}\right)<\infty$ for some $\ell\in\N$, then 
    \begin{eqnarray}
\E[S^{\ell}]&=&\sum_{\pi\in\Pi^\ell(k)}\int_{\mathbb{X}^{|\pi|}}\int_{\mathbb{M}^{|\pi|\times|\pi|}_{\ne}}(f^{\otimes\ell})_{\pi}(\bm{x},\bm{u})\mathbb{Q}^{|\pi|\times|\pi|}_{\ne}(\mathrm{d}\bm{u})\lambda^{|\pi|}(\mathrm{d}\bm{x}),\label{momeq-1}\\
\E[(S-\E S)^{\ell}]&=&\sum_{\pi\in\overline{\Pi}^\ell(k)}\int_{\mathbb{X}^{|\pi|}}\int_{\mathbb{M}^{|\pi|\times|\pi|}_{\ne}}(f^{\otimes\ell})_{\pi}(\bm{x},\bm{u})\mathbb{Q}^{|\pi|\times|\pi|}_{\ne}(\mathrm{d}\bm{u})\lambda^{|\pi|}(\mathrm{d}\bm{x}),\label{momeq-2}\\
\cum_\ell(S)&=&\sum_{\pi\in\widetilde{\Pi}^\ell(k)}\int_{\mathbb{X}^{|\pi|}}\int_{\mathbb{M}^{|\pi|\times|\pi|}_{\ne}}(f^{\otimes\ell})_{\pi}(\bm{x},\bm{u})\mathbb{Q}^{|\pi|\times|\pi|}_{\ne}(\mathrm{d}\bm{u})\lambda^{|\pi|}(\mathrm{d}\bm{x}),\label{momeq-3}
\end{eqnarray}
where $\cum_\ell(S)$ stands for the $\ell$-th cumulant of $S$.
\end{prop}

\subsection{Poisson process approximation}

For $k\in\mathbb{N}$ with $k\geq2$ and a locally compact second countable Hausdorff space $\mathbb{Y}$ with Borel $\sigma$-algebra $\mathcal{Y}$ we consider a measurable function $g: \operatorname{dom} g\to\mathbb{Y}$ with $\operatorname{dom} g\subseteq \mathbb{X}^k\times \mathbb{M}^{k\times k}_{\neq}$. Moreover, we define
$$
{\operatorname{dom}'} g := \left\{( x_1, \ldots, x_k)\in\mathbb{X}^k : \text{ there exists a } \mathbf{u}\in\mathbb{M}^{k\times k}_{\neq} \text{ such that } \left( x_1, \ldots, x_k, \mathbf{u} \right) \in  \operatorname{dom} g\right\}.
$$
In the following, we consider the point process
\begin{equation}\label{definition_xi}
\xi:= \frac{1}{k!}\sum_{(x_1, \ldots, x_k) \in \eta^k_{\neq}\cap  {\operatorname{dom}'} g}  \delta_{g \left( x_1, \ldots, x_k, (M_{x_i,x_j})_{i,j\in[k], i \neq j} \right)}.
\end{equation}
The intensity measure $\mathbf{L}$ of $\xi$ is given by
$$
\mathbf{L}(A):= \frac{1}{k!} \int_{\operatorname{dom} g} \mathbf{1}\{ g(x_1,\hdots,x_k,\mathbf{u})\in A \} \, \lambda^k\otimes \mathbb{Q}_{\neq}^{k\times k}(\mathrm{d} (x_1,\hdots,x_k,\mathbf{u})) 
$$
for $A\in\mathcal{Y}$.

\begin{theorem} \label{Poisson_process_approximation}
Let $\xi$ be the point process given in \eqref{definition_xi} and assume that $\E \xi(\mathbb{Y})^2<\infty$. Let $\zeta$ be a Poisson process with finite intensity measure $\mathbf{M}$. Then,
    \begin{align*}
        d_{KR}({\xi}, \zeta) & \leq d_{TV}(\mathbf{L},\mathbf{M}) + 2 \big( \E \xi(\mathbb{Y})^2 - (\E \xi(\mathbb{Y}))^2 - \E \xi(\mathbb{Y}) \big) \\
        & \leq d_{TV}(\mathbf{L},\mathbf{M}) + \frac{2^{k+1}}{k!} \, r(\operatorname{dom} g),
    \end{align*}
where
\begin{align*}
  &  r(\operatorname{dom} g) := \max_{1 \leq \ell \leq k-1} 
\int_{\mathbb{X}^{2k-\ell}} \int_{\mathbb{M}^{(2k-\ell)\times (2k-\ell)}_{\neq}}  
\mathbf{1}\big\{ ( {x}_1, \ldots, {x}_k , (\mathbf{u}_{i,j})_{i,j\in[k],i\neq j} ) \in \operatorname{dom} g  \big\}  \\
&  \hspace{3cm} \times \mathbf{1}\big\{ ( {x}_1, \ldots, {x}_\ell, {x}_{k+1}, \ldots, {x}_{2k-\ell}, (\mathbf{u}_{i,j})_{i,j\in [\ell]\cup\{k+1,\hdots,2k-\ell\}, i\neq j} ) \in \operatorname{dom} g \big\} \\
& \hspace{3cm} \times  \mathbb{Q}^{(2k-\ell)\times(2k-\ell)}_{\neq} (d\mathbf{u}) \, \lambda^{2k-\ell}(d(x_1,\hdots,x_{2k-\ell})).
\end{align*}  
\end{theorem}

Next we provide some corollaries of Theorem \ref{Poisson_process_approximation}. For $g: \operatorname{dom} g \to\mathbb{Y}$ as above and $B\in\mathcal{Y}$ we define the $U$-statistic
$$
T:=\frac{1}{k!}\sum_{(x_1,\hdots,x_k)\in\eta^k_{\neq}\cap \operatorname{dom}'g} \mathbf{1}\{g(x_1,\hdots,x_k,(M_{x_i,x_j})_{i,j\in[k],i\neq j})\in B\}.
$$

\begin{cor}
Assume that $\mathbb{E}[T^2]<\infty$ and let $Z$ be a Poisson distributed random variable with parameter $a\in[0,\infty)$. Then,
\begin{align*}
d_{\operatorname{Wass}}(T,Z) & \leq |\mathbb{E}[T]-a| + 2 ( \mathbb{E}[T^2]- \mathbb{E}[T] -(\mathbb{E}[T])^2 ) \\
& \leq |\mathbb{E}[T]-a| + \frac{2^{k+1}}{k!} r(B)
\end{align*}
with
\begin{align*}
  &  r(B) := \max_{1 \leq \ell \leq k-1} 
\int_{\mathbb{X}^{2k-\ell}} \int_{\mathbb{M}^{(2k-\ell)\times (2k-\ell)}_{\neq}}  
\mathbf{1}\big\{ g({x}_1, \ldots, {x}_k , (\mathbf{u}_{i,j})_{i,j\in[k],i\neq j} ) \in B \big\}  \\
&  \hspace{3cm} \times \mathbf{1}\big\{ g({x}_1, \ldots, {x}_\ell, {x}_{k+1}, \ldots, {x}_{2k-\ell}, (\mathbf{u}_{i,j})_{i,j\in [\ell]\cup\{k+1,\hdots,2k-\ell\}, i\neq j} ) \in B \big\} \\
& \hspace{3cm} \times  \mathbb{Q}^{(2k-\ell)\times(2k-\ell)}_{\neq} (d\mathbf{u}) \, \lambda^{2k-\ell}(d(x_1,\hdots,x_{2k-\ell})).
\end{align*}  
\end{cor}

\begin{proof}
The statement follows from Theorem \ref{Poisson_process_approximation} exactly as Theorem 7.1 follows from Theorem 3.1 in \cite{dst}.
\end{proof}

Let $g: \mathbb{X}^k\times\mathbb{M}^{k\times k}\to\mathbb{R}$ be a measurable function. Define the $U$-statistic
\begin{equation}\label{definition_U}
U := \frac{1}{k!}\sum_{(x_1,\hdots,x_k)\in\eta^k_{\neq}\cap \operatorname{dom}'g} g(x_1,\hdots,x_k,(M_{x_i,x_j})_{i,j\in[k],i\neq j}). 
\end{equation}
Moreover, let
\begin{align*}
\mathbf{L}(A) := \frac{1}{k!} \, \mathbb{E} \sum_{(x_1, \ldots, x_k) \in \eta_{\neq}^k} 
\mathbf{1}\left( g(x_1, \ldots, x_k, (M_{x_i,x_j})_{i,j\in[k], i \neq j}) \in A \setminus \{0\} \right)
\end{align*}
for $A \in \mathcal{B}(\mathbb{R})$. We get the following result.

\begin{cor}\label{compound}
For a Poisson process  \( \zeta \) on \( \mathbb{R} \) with a finite intensity measure \( \mathbf{M} \) let
\[
Y := \sum_{x \in \zeta} x.
\]
Then,
\begin{align*}
d_{\mathrm{TV}}(U, Y) \le d_{\mathrm{TV}}(\mathbf{L}, \mathbf{M}) + \frac{2^{k+1}}{k!} \, r_0
\end{align*}
with
\begin{align*}
  &  r_0 := \max_{1 \leq \ell \leq k-1} 
\int_{\mathbb{X}^{2k-\ell}} \int_{\mathbb{M}^{(2k-\ell)\times (2k-\ell)}_{\neq}}  
\mathbf{1}\big\{ g({x}_1, \ldots, {x}_k , (\mathbf{u}_{i,j})_{i,j\in[k],i\neq j} )\neq 0 \big\}  \\
&  \hspace{3cm} \times \mathbf{1}\big\{ g({x}_1, \ldots, {x}_\ell, {x}_{k+1}, \ldots, {x}_{2k-\ell}, (\mathbf{u}_{i,j})_{i,j\in [\ell]\cup\{k+1,\hdots,2k-\ell\}, i\neq j} ) \neq 0 \big\} \\
& \hspace{3cm} \times  \mathbb{Q}^{(2k-\ell)\times(2k-\ell)}_{\neq} (d\mathbf{u}) \, \lambda^{2k-\ell}(d(x_1,\hdots,x_{2k-\ell})).
\end{align*}
\end{cor}

\begin{proof}
The assertion can be deduced from Theorem \ref{Poisson_process_approximation} as Theorem 7.4 is derived from Theorem 3.1 in \cite{dst}.
\end{proof}

Next we compare the distribution of the random variable $U$ introduced in \eqref{definition_U} with an $\alpha$-stable distribution. To this end we fix $\alpha\in(0,1)$ and let
$$
g_\alpha (u) := \mathbf{1}\{ u\neq 0 \} \operatorname{sign}(g(u)) g(u)^{-\alpha}
$$
for $u\in\mathbb{R}$, where $\operatorname{sign}(u) = \mathbf{1}(u \ge 0) - \mathbf{1}(u < 0)$.
We define
\begin{align*}
\mathbf{L}^{(\alpha)}(A) & := \frac{1}{k!} \, \mathbb{E} \sum_{(x_1, \ldots, x_k) \in \eta_{\neq}^k}  \mathbf{1}\{ g_\alpha(x_1, \ldots, x_k, (M_{x_i,x_j})_{i,j\in[k], i \neq j})\in A\setminus\{0\} \}
\end{align*}
for $A \in \mathcal{B}(\mathbb{R})$.

\begin{cor}\label{approximation_alpha-stable}
Let \( \alpha \in (0, 1) \), and let \( \mathbf{M} \) be either the Lebesgue measure on \( \mathbb{R} \) or its restriction to \( (0,\infty)) \). Define
\[
Y := \sum_{x \in \zeta} \operatorname{sign}(x) |x|^{-1/\alpha},
\]
where \( \zeta \) is a Poisson process with intensity measure \( \mathbf{M}\). Assume that there exist functions \( g_1, g_2, g_3 \colon (0,\infty) \to (0,\infty) \) such that, for every \( a > 0 \),
\begin{align*}
d_{\mathrm{TV}}(\mathbf{L}|_{[-a,a]}, \mathbf{M}|_{[-a,a]}) &\le g_1(a),
\end{align*}
\begin{align*}
&\max_{1 \leq \ell \leq k-1} 
\int_{\mathbb{X}^{2k-\ell}} \int_{\mathbb{M}^{(2k-\ell)\times (2k-\ell)}_{\neq}}  
\mathbf{1}\big\{ g_\alpha({x}_1, \ldots, {x}_k , (\mathbf{u}_{i,j})_{i,j\in[k],i\neq j} )\in A\setminus\{0\} \big\}  \\
&  \hspace{3cm} \times \mathbf{1}\big\{ g_\alpha({x}_1, \ldots, {x}_\ell, {x}_{k+1}, \ldots, {x}_{2k-\ell}, (\mathbf{u}_{i,j})_{i,j\in [\ell]\cup\{k+1,\hdots,2k-\ell\}, i\neq j} ) \in A\setminus\{0\} \big\} \\

&\le g_2(a),
\end{align*}
\begin{align*}

\end{align*}
Then there exists a constant \( C > 0 \), depending only on \( \alpha \) and \( k \), such that
\[

\]
where
\[
b := \inf_{a > 0} \max\left\{ a^{1/2 - 1/(2\alpha)},\; g_1(a),\; g_2(a),\; \sqrt{g_3(a)} \right\}.
\]
\end{cor}

\begin{proof}
The proof is analogously to the derivation of Theorem 7.8 from Theorem 3.1 in \cite{dst}. 
\end{proof}

We can employ Theorem~\ref{Poisson_process_approximation} %\cite[Theorem 7.1]{dst} 
in order to bound
$$d_W (\hat{S}_t,Z) $$
for a Poisson distributed random variable $Z$ with mean $\lambda\in[0,\infty)$. By the latter theorem, we have

\[
d_W(\hat{S}, Z) \leq \left| \mathbb{E}[\hat{S}] - \lambda \right| + \frac{2^{k+1}}{k!} \, r_t(B)
\]
for
\begin{align*}
    r_t(B) := \max_{1 \leq \ell \leq k-1} 
\int_{\hat{X}^\ell} & \left( 
\int_{\hat{X}^{k - \ell}} 
\mathbf{1}\big( \hat{f}_t ( \hat{x}_1, \ldots, \hat{x}_k ) \in B \big) \, 
\hat{K}_t^{k - \ell}(d(x_{\ell+1}, \ldots, x_k))

& \times \hat{K}_t^\ell(d(x_1, \ldots, x_\ell)),
\end{align*}
where $\hat{K}_t = t \hat{K}$ denotes the intensity measure of the Poisson point process $\hat{\eta}$ (and $\hat{K}$ is a $\sigma$-finite measure on the mark space). We write
\begin{align*}

\int_{\hat{X}^{k - \ell}} 
\mathbf{1}\big( \hat{f}_t ( \hat{x}_1, \ldots, \hat{x}_k ) \in B \big) \, 
\hat{K}_t^{k - \ell}(d(\hat{x}_{\ell+1}, \ldots, \hat{x}_k))

 & + \int_{X^\ell}  
\int_{\hat{X}^{k - \ell}}\int_{\hat{X}^{k - \ell}} 
\mathbf{1}\big( \hat{f}_t ( \hat{x}_1, \ldots, \hat{x}_k ) \in B \big) \,  \mathbf{1}\big( \hat{f} _t( \hat{x}_1, \ldots, \hat{x}_\ell, \hat{x}'_{\ell+1} \ldots, \hat{x}'_{k} ) \in B \big) \\

\int_{X^{k - \ell}}\int_{X^{k - \ell}} 
\mathbf{1}\big(  ( {x}_1, \ldots, {x}_k, {x}'_{\ell+1} \ldots, {x}'_{k} ) \notin \Delta_{2k-\ell} \big) \,  \\
&  \times \mathbf{1}\big( {f}_t ( {x}_1, \ldots, {x}_k , w ) \in B \big)\mathbf{1}\big( {f}_t ( {x}_1, \ldots, {x}_\ell, {x}'_{\ell+1} \ldots, {x}'_{k}, w ) \in B \big) \\

\end{align*}

\begin{theorem} \label{P1}
    Let $S$ be the U-statistics given in \eqref{originalUstat}. Let $Z$ be a Poisson distributed random variable with mean $\lambda$. Assume either that
    \begin{itemize}
        \item[(i)] the intensity measure $K_t \otimes \mathbb{Q}$ is finite
    \end{itemize}
    or
    \begin{itemize}
        \item[(ii)] $\mathbb{P} (\exists r_0>0 \,: \, \hat{\eta}( \mathbb{R}^d \times \cdot) = \hat{\eta}( (B(0,r) \times \cdot) \, \forall r \geq r_0 ) = 1$.
    \end{itemize}
    Then it holds
    \begin{align*}
        d_W({S}, Z) \leq \left| \mathbb{E}[{S}] - \lambda \right| + \frac{2^{k+1}}{k!} \, s_t(B)
    \end{align*}
for
\begin{align*}
  &  s_t(B) := \max_{1 \leq \ell \leq k-1} 
\int \int_{X^\ell}  
\int_{X^{k - \ell}}\int_{X^{k - \ell}} \mathbf{1}\big( {f}_t ( {x}_1, \ldots, {x}_k , w ) \in B \big)  \\

&  \times  {K}_t^{k - \ell}(d({x}'_{\ell+1}, \ldots, {x}'_k)) {K}_t^\ell(d({x}_1, \ldots, {x}_\ell)) d\mathbb{Q}^{\frac{(2k - \ell) (2k -\ell-1)}{2}} (dw) .
\end{align*}
  
\end{theorem}




\subsubsection{Compound Poisson approximation}

As another consequence of Theorem \ref{Poisson_process_approximation} we get the analogue of \cite[Theorem 7.4]{dst}. Consider the family of U-statistics

\begin{align*}
S_t := \frac{1}{k!} \sum_{(x_1, \ldots, x_k) \in \eta_{t,\neq}^k} h_t(x_1, \ldots, x_k, (W_{x_i,x_j})_{i,j =1,\ldots, k, i \neq j}), \quad t \ge 1,
\end{align*}
for a symmetric function $h_t$. Then define

\begin{align*}
L_t(A) := \frac{1}{k!} \, \mathbb{E} \sum_{(x_1, \ldots, x_k) \in \eta_{t, \neq}^k} 
\mathbf{1}\left( h_t(x_1, \ldots, x_k, (W_{x_i,x_j})_{i,j =1,\ldots, k, i \neq j}) \in t^{-\gamma} A \setminus \{0\} \right),
\end{align*}
for $A \in \mathcal{B}(\mathbb{R})$. We get the following result.

\begin{theorem}\label{compound}
Let \( \zeta \) be a Poisson process on \( \mathbb{R} \) with a finite intensity measure \( M \), let
\[
Z := \sum_{x \in \zeta} x,
\]
and let \( \gamma \in \mathbb{R} \). Then, for \( t \ge 1 \),
\begin{align*}
d_{\mathrm{TV}}(t^{\gamma} S_t, Z) \le d_{\mathrm{TV}}(L_t, M) + \frac{2^{k+1}}{k!} \, r_t,
\end{align*}
with
\begin{align*}
  &  r_t(A) := \max_{1 \leq \ell \leq k-1} 
\int \int_{X^\ell}  
\int_{X^{k - \ell}}\int_{X^{k - \ell}} \\
& \, \mathbf{1}\big(  h_t( {x}_1, \ldots, {x}_k , w ) \neq 0  \big) \mathbf{1}\big( h_t( {x}_1, \ldots, {x}_\ell, {x}'_{\ell+1} \ldots, {x}'_{k}, w ) \neq 0 \big) \\
&\,{K}^{k - \ell}(d({x}_{\ell+1}, \ldots, {x}_k))   {K}^{k - \ell}(d({x}'_{\ell+1}, \ldots, {x}'_k)) {K}^\ell(d({x}_1, \ldots, {x}_\ell)) d\mathbb{Q}^{\frac{(2k - \ell) (2k -\ell-1)}{2}} (dw) .
\end{align*}
\end{theorem}








\subsubsection{Approximation by $\alpha$-stable distribution}

Fix $\alpha \in (0, 1)$ and $\gamma \in \mathbb{R}$. The goal is to make use of a functional limit theorem for the point processes
\begin{align*}
    t^\gamma \cdot \xi_t &:= \frac{1}{k!} \sum_{(x_1, \ldots, x_k) \in \eta^k_{\neq}} 
\mathbf{1}(h_t(x_1, \ldots, x_k,(W_{x_i,x_j})_{i,j =1,\ldots, k, i \neq j}) \ne 0) \\
&\quad \delta_{\operatorname{sign}(h_t(x_1, \ldots, x_k,(W_{x_i,x_j})_{i,j =1,\ldots, k, i \neq j})) t^\gamma |h_t(x_1, \ldots, x_k,(W_{x_i,x_j})_{i,j =1,\ldots, k, i \neq j})|^{-\alpha}},
\end{align*}
 $t \ge 1$, on $\mathbb{R}$, with $\operatorname{sign}(a) = \mathbf{1}(a \ge 0) - \mathbf{1}(a < 0)$. In fact, having established Theorem~\ref{Poisson_process_approximation} the following application follows from the proof of \cite[Theorem 7.8]{dst}. In the following we denote by $L_t$ the intensity measure of $t^\gamma \cdot \xi_t$, and

\begin{align*}
  &  r_t(A) := \max_{1 \leq \ell \leq k-1} 
\int \int_{X^\ell}  
\int_{X^{k - \ell}}\int_{X^{k - \ell}} \\
& \quad \mathbf{1}\big(  h_t( {x}_1, \ldots, {x}_k , w ) \neq 0  \big) \mathbf{1}\left( \operatorname{sign}(h_t(x_1, \ldots, x_k,w)) \, |h_t(x_1, \ldots, x_k,w)|^{-\alpha} \in t^{-\gamma} A \right)
 \\
&  \times \mathbf{1}\big( h_t( {x}_1, \ldots, {x}_\ell, {x}'_{\ell+1} \ldots, {x}'_{k}, w ) \neq 0 \big) \\
& \times \mathbf{1}\left( \operatorname{sign}(h_t({x}_1, \ldots, {x}_\ell, {x}'_{\ell+1} \ldots, {x}'_{k},w)) \, |h_t({x}_1, \ldots, {x}_\ell, {x}'_{\ell+1} \ldots, {x}'_{k},w)|^{-\alpha} \in t^{-\gamma} A \right)\\
&\,{K}^{k - \ell}(d({x}_{\ell+1}, \ldots, {x}_k))   {K}^{k - \ell}(d({x}'_{\ell+1}, \ldots, {x}'_k)) {K}^\ell(d({x}_1, \ldots, {x}_\ell)) d\mathbb{Q}^{\frac{(2k - \ell) (2k -\ell-1)}{2}} (dw) .
\end{align*}

\begin{theorem}\label{stable}
Let \( \alpha \in (0, 1) \), and let \( M \) be either the Lebesgue measure on \( \mathbb{R} \) or its restriction to \( \mathbb{R}_+ \). Define
\[
Z := \sum_{x \in \zeta} \operatorname{sign}(x) |x|^{-1/\alpha},
\]
where \( \zeta \) is a Poisson process with intensity measure \( M\). Assume that there exist a constant \( \gamma \in \mathbb{R} \) and functions \( g_1, g_2, g_3 \colon \mathbb{R}_+^2 \to \mathbb{R}_+ \) such that, for every \( a > 0 \) and \( t \ge 1 \),
\begin{align*}
d_{\mathrm{TV}}(L_t|_{[-a,a]}, M|_{[-a,a]}) &\le g_1(a, t), \\
r_t([-a,a]) &\le g_2(a, t), \\
\frac{t^{-\gamma/\alpha}}{k!} \, \mathbb{E} \sum_{(x_1, \ldots, x_k) \in \eta^k_{\neq}}

&\le g_3(a, t).
\end{align*}
Then there exists a constant \( C > 0 \), depending only on \( \alpha \) and \( k \), such that
\[

\]
where

\]
\end{theorem}

\begin{lemma}\label{lma1}

Then,
\begin{align*}
& \lim_{n\to\infty} \int_{\widehat{\mathbb{X}}^m} h(x_1,\hdots,x_m, \tilde{m}^{(n)}(\hat{x}_1,\hdots,\hat{x}_m)) \, \widehat{\lambda}^m(d(\hat{x}_1,\hdots,\hat{x}_m) ) \\
& = \int_{\mathbb{X}^m} \int_{\mathbb{M}^{m\times m}_{\neq}} h(x_1,\hdots,x_m, u) \, \mathbb{Q}_{\neq}^{m\times m}(du) \, \lambda^m(d(x_1,\hdots,x_m)).
\end{align*}
\end{lemma}

\begin{proof}
Define
$$
\Delta^{(n)}_m := \{ (x_1,\hdots,x_m) \in \mathbb{X}^m: \exists~\ell\in\mathbb{N}, i,j\in[m], i\ne j\text{ with $x_i,x_j\in A^{(n)}_{\ell}$}\}
$$
and
$$
\widehat{\Delta}^{(n)}_m := \{ (\hat{x}_1,\hdots,\hat{x}_m) \in \widehat{\mathbb{X}}^m:  \exists~\ell\in\mathbb{N}, i,j\in[m], i\ne j\text{ with $x_i,x_j\in A^{(n)}_{\ell}$}\}.
$$
For $(\hat{x}_1,\hdots,\hat{x}_m)\in\widehat{\mathbb{X}}^m\backslash\widehat{\Delta}_m^{(n)}$, $\tilde{m}^{(n)}$ assigns different marks to each pair $(i,j)$ with $i,j\in[m]$ and $i\neq j$ so that
\begin{align*}
& \int_{\widehat{\mathbb{X}}^m\backslash\widehat{\Delta}_m^{(n)}} h(x_1,\hdots,x_m, \tilde{m}^{(n)}(\hat{x}_1,\hdots,\hat{x}_m)) \, \widehat{\lambda}^m(d(\hat{x}_1,\hdots,\hat{x}_m) ) \\
& = \int_{\mathbb{X}^m\backslash\Delta_m^{(n)}} \int_{\mathbb{M}^{m\times m}_{\neq}} h(x_1,\hdots,x_m, u) \, \mathbb{Q}_{\neq}^{m\times m}(du) \, \lambda^m(d(x_1,\hdots,x_m)).
\end{align*}
It follows from \eqref{eqn:monotonicity_partition} and \eqref{eqn:size_partition} that
$$
\mathbf{1}\{ (\hat{x}_1,\hdots,\hat{x}_m)\in\widehat{\mathbb{X}}^m\backslash\widehat{\Delta}_m^{(n)}\} \to 1 \quad \text{as} \quad n\to \infty \quad \text{for} \quad \widehat{\lambda}^m\text{-a.e. } (\hat{x}_1,\hdots,\hat{x}_m)\in\widehat{\mathbb{X}}^m
$$
and
$$
\mathbf{1}\{ (x_1,\hdots,x_m)\in\mathbb{X}^m\backslash\Delta_m^{(n)}\} \to 1 \quad \text{as} \quad n\to \infty \quad \text{for} \quad \lambda^m\text{-a.e. } (x_1,\hdots,x_m)\in\mathbb{X}^m.
$$
Due to the boundedness of $h$ and the assumptions on its support we can apply the dominated convergence theorem and obtain
\begin{align*}
& \lim_{n\to\infty}\int_{\widehat{\mathbb{X}}^m\backslash\widehat{\Delta}_m^{(n)}} h(x_1,\hdots,x_m, \tilde{m}^{(n)}(\hat{x}_1,\hdots,\hat{x}_m)) \, \widehat{\lambda}^m(d(\hat{x}_1,\hdots,\hat{x}_m) ) \\
& = \lim_{n\to\infty}\int_{\mathbb{X}^m\backslash\Delta_m^{(n)}} \int_{\mathbb{M}^{m\times m}_{\neq}} h(x_1,\hdots,x_m, u) \, \mathbb{Q}_{\neq}^{m\times m}(du) \, \lambda^m(d(x_1,\hdots,x_m)) \\ & = \int_{\mathbb{X}^m} \int_{\mathbb{M}^{m\times m}_{\neq}} h(x_1,\hdots,x_m, u) \, \mathbb{Q}_{\neq}^{m\times m}(du) \, \lambda^m(d(x_1,\hdots,x_m))
\end{align*}
and
$$
\lim_{n\to\infty} \int_{ \widehat{\Delta}^{(n)}_m} h(x_1,\hdots,x_m, \tilde{m}^{(n)}(\hat{x}_1,\hdots,\hat{x}_m)) \, \widehat{\lambda}^m(d(\hat{x}_1,\hdots,\hat{x}_m) )=0.
$$
Combining the two limits proves the statement.
\end{proof}

\begin{lemma}\label{lem:limits_S_n}
    Let $f: \mathbb{X}^k\times \mathbb{M}^{k\times k}_{\neq}\to\mathbb{R}$ be a measurable function satisfying that $\supp(f)\subseteq B^k\times\mathbb{M}^{k\times k}_{\ne}$ with $\lambda(B)<\infty$ and $\|f\|_{\infty}<\infty$.  
    \begin{enumerate}
        \item[1)] For $p>0$, we have $\E\left[\left|S-\widehat{S}^{(n)}\right|^p\right]\to0$, as $n\to\infty$.
        \item[2)] For $\ell\in\N$, we have 
       \begin{equation} \lim_{n\to\infty}\E\left[(\widehat{S}^{(n)})^{\ell}\right]=\E\left[S^{\ell}\right],
       \end{equation}
       and 
       \begin{eqnarray}
        \E[S^{\ell}]=\sum_{\pi\in\Pi^\ell(k)}\int_{\mathbb{X}^{|\pi|}}\int_{\mathbb{M}^{|\pi|\times|\pi|}_{\ne}}(f^{\otimes\ell})_{\pi}(\bm{x},\bm{u})\mathbb{Q}^{|\pi|\times|\pi|}_{\ne}(\mathrm{d}\bm{u})\lambda^{|\pi|}(\mathrm{d}\bm{x}).
       \end{eqnarray}
    \end{enumerate}
\end{lemma}
\begin{proof}
\begin{enumerate}
        \item[1)] 
According to the coupling $(\widetilde{S}^{(n)},\widehat{S}^{(n)})$ constructed in previous section, together with the Cauchy-Schwarz inequality and \eqref{diff2} it follows that 
\begin{eqnarray*}
    &&\E\left[\left|S-\widehat{S}^{(n)}\right|^p\right]\\
    &=&\E\left[\left|\widetilde{S}^{(n)}-\widehat{S}^{(n)}\right|^p\mathbf{1}_{\{\widetilde{S}^{(n)}\ne\widehat{S}^{(n)}\}}\right]\\
    &=&\E\left[\left|\sum_{(x_1,\hdots,x_k)\in(\eta_{\mathbb{X}})^k_{\neq}} f(x_1,\hdots,x_k, (\widetilde{M}^{(n)}_{x_i,x_j})_{i,j\in[k],i\neq j})-\widehat{S}^{(n)}\right|^p\mathbf{1}_{\{\widetilde{S}^{(n)}\ne\widehat{S}^{(n)}\}}\right]\\
    &=&\E\left[\left|\sum_{(\hat{x}_1,\hdots,\hat{x}_k)\in\widehat{\eta}^k_{\neq}} f(x_1,\hdots,x_k, (\widetilde{M}^{(n)}_{x_i,x_j})_{i,j\in[k],i\neq j})-\widehat{S}^{(n)}\right|^p\mathbf{1}_{\{\widetilde{S}^{(n)}\ne\widehat{S}^{(n)}\}}\right]\\
    &=&\E\left[\left|\sum_{(\hat{x}_1,\hdots,\hat{x}_k)\in\widehat{\eta}^k_{\neq}} \left[f(x_1,\hdots,x_k, (\widetilde{M}^{(n)}_{x_i,x_j})_{i,j\in[k],i\neq j})-f(x_1,\hdots,x_k, \tilde{m}^{(n)}(\hat{x}_1,\hdots,\hat{x}_k))\right]\right|^p\mathbf{1}_{\{\widetilde{S}^{(n)}\ne\widehat{S}^{(n)}\}}\right]\\

\end{eqnarray*}
Therefore, for $p>0$, we have $\lim_{n\to0}\E\left[\left|S-\widehat{S}^{(n)}\right|^p\right]=0$.
\item[2)]
For $\ell\in\N$, $\lim_{n\to0}\E\left[\left|S-\widehat{S}^{(n)}\right|^{\ell}\right]=0$ implies $$\E\left[S^{\ell}\right]=\lim_{n\to\infty}\E\left[(\widehat{S}^{(n)})^{\ell}\right].$$ On the other hand, we have

\end{proof}
\begin{lemma}\label{lem:limits_S_n2}
   Let $f: \mathbb{X}^k\times \mathbb{M}^{k\times k}_{\neq}\to\mathbb{R}$ be a measurable function. 
\begin{enumerate}
\item[1)] Suppose $f\ge0$ and $\E[S^{\ell}]<\infty$ for some $\ell\in\N$, then there exists a sequence of functions $(f_m)_{m\in\N}$ such that 
\begin{equation}
    \lim_{m\to\infty}\E\left[\left|S-S_m\right|^{\ell}\right]=0,
\end{equation}
with 
$$S_m:=\sum_{(x_1,\hdots,x_k)\in\eta^k_{\neq}} f_m(x_1,\hdots,x_k, (M_{x_i,x_j})_{i,j\in[k], i\neq j}),$$
and 
\begin{equation}
\E[S^{\ell}]=\sum_{\pi\in\Pi^\ell(k)}\int_{\mathbb{X}^{|\pi|}}\int_{\mathbb{M}^{|\pi|\times|\pi|}_{\ne}}(f^{\otimes\ell})_{\pi}(\bm{x},\bm{u})\mathbb{Q}^{|\pi|\times|\pi|}_{\ne}(\mathrm{d}\bm{u})\lambda^{|\pi|}(\mathrm{d}\bm{x}).
\end{equation}
\item[2)]Suppose that $\E\left(\overline{S}^{\ell}\right)<\infty$ for some $\ell\in\N$ with $\overline{S}$ given in \eqref{def-integ1}. Then, the above results prevail. 
\end{enumerate}
\end{lemma}
\begin{proof}
Because $\lambda$ is $\sigma$-finite, 
there exists an increasing sequence of measurable sets $(B_m)_{m\in\mathbb{N}}$ such that $\lambda(B_m)<\infty$ for each $m\in\mathbb{N}$ and $ \cup_{m=1}^\infty B_m=\mathbb{X}$. Take $f_m:\mathbb{X}^k\times \mathbb{M}^{k\times k}_{\neq}\to\mathbb{R}$ defined by 
\begin{equation*}
f_m(x_1,\hdots,x_k, (u_{i,j})_{i,j\in[k], i\neq j}):=f(x_1,\hdots,x_k, (u_{i,j})_{i,j\in[k], i\neq j})\mathbf{1}_{\{(x_1,\hdots,x_k)\in B_m\}}\mathbf{1}_{\{|f|\le m\}}.
\end{equation*}
It is clear that $f_m$ converges to $f$ as $m\to\infty$ pointwisely.
Let $S_m$ be defined by replacing $f$ with $f_m$ in \eqref{def:ustat}. 
\begin{enumerate}
\item[1)] 
From Lemma~\ref{lem:limits_S_n}, we have 
\begin{equation}
    \E[(S_m)^\ell]=\sum_{\pi\in\Pi^\ell(k)}\int_{\mathbb{X}^{|\pi|}}\int_{\mathbb{M}^{|\pi|\times|\pi|}_{\ne}}(f_m^{\otimes\ell})_{\pi}(\bm{x},\bm{u})\mathbb{Q}^{|\pi|\times|\pi|}_{\ne}(\mathrm{d}\bm{u})\lambda^{|\pi|}(\mathrm{d}\bm{x}).
\end{equation}
When $f\ge0$, it gives $0\le f-f_m\le f$. Therefore, by the dominated convergence theorem, we have $\lim_{m\to\infty}\E\left[\left|S-S_m\right|^{\ell}\right]=0$. It further provides that
\begin{eqnarray*}
    \E[S^\ell]&=&\lim_{m\to\infty}\E[(S_m)^\ell]\\
    &=&\lim_{m\to\infty}\sum_{\pi\in\Pi^\ell(k)}\int_{\mathbb{X}^{|\pi|}}\int_{\mathbb{M}^{|\pi|\times|\pi|}_{\ne}}(f_m^{\otimes\ell})_{\pi}(\bm{x},\bm{u})\mathbb{Q}^{|\pi|\times|\pi|}_{\ne}(\mathrm{d}\bm{u})\lambda^{|\pi|}(\mathrm{d}\bm{x})\\
    &=&\sum_{\pi\in\Pi^\ell(k)}\int_{\mathbb{X}^{|\pi|}}\int_{\mathbb{M}^{|\pi|\times|\pi|}_{\ne}}(f^{\otimes\ell})_{\pi}(\bm{x},\bm{u})\mathbb{Q}^{|\pi|\times|\pi|}_{\ne}(\mathrm{d}\bm{u})\lambda^{|\pi|}(\mathrm{d}\bm{x}).
\end{eqnarray*}
\item[2)] For general $f$ such that $\E\left(\overline{S}^{\ell}\right)<\infty$, we can write $f=f^+-f^-$ where $f^+$ and $f^-$ are defined by 
$$f^+:=f\mathbf{1}_{\{f\ge0\}},~~~~f^-:=-f\mathbf{1}_{\{f<0\}}.$$
Similarly, we denote $f_m=f_m^+-f_m^-$, and define $S^+$, $S^-$, $S^+_m$ and $S_m^-$ by replacing $f$ in \eqref{def:ustat} by $f^+$, $f^-$, $f_m^+$ and $f_m^-$ respectively.
For $\ell\in\N$, by Minkowski inequality, 
\begin{eqnarray*}
    \E\left[\left|S-S_m\right|^{\ell}\right]&=&\E\left[\left|(S^+-S^-)-(S_m^+-S_m^-)\right|^{\ell}\right]\\
    &=&\E\left[\left|(S^+-S_m^+)-(S^--S_m^-)\right|^{\ell}\right]\\

\end{eqnarray*}

\end{enumerate}
Moreover, using Lemma~\ref{lem:limits_S_n}, we have 
\begin{equation}

\end{equation}
\end{proof}

\subsection{Proof of Theorem \ref{Poisson_process_approximation}}

\begin{proof}[Proof of Theorem \ref{Poisson_process_approximation}]
The second inequality is an immediate consequence from Lemma \ref{lem:limits_S_n2} and \eqref{eqn:generalised_Mecke_U-statistics}, which yield that
\begin{align*}
& \E\xi(\mathbb{Y})^2 - (\E\xi(\mathbb{Y}))^2 - \E\xi(\mathbb{Y})  \\
& = \Var\xi(\mathbb{Y}) - \E\xi(\mathbb{Y})  \\
& = \sum_{\ell=1}^k \frac{\ell!}{(k!)^2} \binom{k}{\ell}^2 \int_{\mathbb{X}^{2k-\ell}} \int_{\mathbb{M}_{\neq}^{(2k-\ell)\times(2k-\ell)}} \mathbf{1}\{(x_1,\hdots,x_k,(\mathbf{u})_{i,j})_{i,j\in[k],i\neq j})\in\operatorname{dom} g\}  \\
& \hspace{4cm} \times \mathbf{1}\{(x_1,\hdots,x_\ell,x_{k+1},\hdots,x_{2k-\ell},(\mathbf{u})_{i,j})_{i,j\in[\ell]\cup\{k+1,\hdots,2k-\ell\}, i\neq j})\in\operatorname{dom} g\} \\
& \hspace{4cm} \times \mathbb{Q}_{\neq}^{(2k-\ell)\times (2k-\ell)}(d\mathbf{u}) \, \lambda^{2k-\ell}(d(x_1,\hdots,x_{2k-\ell})) \\
& \quad - \frac{1}{k!} \int_{\mathbb{X}^k} \int_{\mathbb{M}_{\neq}^{k\times k}} \mathbf{1}\{(x_1,\hdots,x_k,(\mathbf{u})_{i,j})_{i,j\in[k],i\neq j})\in\operatorname{dom} g\} \, \mathbb{Q}_{\neq}^{k\times k}(d\mathbf{u}) \, \lambda^{k}(d(x_1,\hdots,x_k)) \\
& = \sum_{\ell=1}^{k-1} \frac{\ell!}{(k!)^2} \binom{k}{\ell}^2 \int_{\mathbb{X}^{2k-\ell}} \int_{\mathbb{M}_{\neq}^{(2k-\ell)\times(2k-\ell)}} \mathbf{1}\{(x_1,\hdots,x_k,(\mathbf{u})_{i,j})_{i,j\in[k],i\neq j})\in\operatorname{dom} g\}  \\
& \hspace{4cm} \times \mathbf{1}\{(x_1,\hdots,x_\ell,x_{k+1},\hdots,x_{2k-\ell},(\mathbf{u})_{i,j})_{i,j\in[\ell]\cup\{k+1,\hdots,2k-\ell\}, i\neq j})\in\operatorname{dom} g\} \\
& \hspace{4cm} \times \mathbb{Q}_{\neq}^{(2k-\ell)\times (2k-\ell)}(d\mathbf{u}) \, \lambda^{2k-\ell}(d(x_1,\hdots,x_{2k-\ell})) \\
& \leq \frac{1}{k!} \sum_{\ell=1}^{k-1} \frac{1}{(k-\ell)!} \binom{k}{\ell} r(\operatorname{dom} g) \leq \frac{2^k}{k!} \sum_{\ell=1}^{k-1} r(\operatorname{dom} g).
\end{align*}

Next we show the first inequality for the case that there exists a set $B\subseteq \mathbb{X}$ such that $\operatorname{dom}' g\subseteq B^k$ and $\lambda(B)<\infty$. For $n\in\mathbb{N}$ let $\widetilde{\xi}^{(n)}$ and $\widehat{\xi}^{(n)}$ be as in Section \ref{Poisson_process_approximation} and recall that $\xi\overset{d}{=}\widetilde{\xi}^{(n)}$. The triangle inequality for the Kantorovich-Rubinstein distance yields that
$$
d_{KR}(\xi,\zeta)=d_{KR}(\widetilde{\xi}^{(n)},\zeta) \leq d_{KR}(\widetilde{\xi}^{(n)},\widehat{\xi}^{(n)}) + d_{KR}(\widehat{\xi}^{(n)},\zeta).
$$
From the definition of the Kantorovich-Rubinstein distance, the Cauchy-Schwarz inequality and \eqref{diff2} it follows that
\begin{align*}
d_{KR}(\widetilde{\xi}^{(n)},\widehat{\xi}^{(n)}) & \leq \E\big[ \mathbf{1}\{ \widetilde{\xi}^{(n)}\neq\widehat{\xi}^{(n)} \} (\widetilde{\xi}^{(n)}(\mathbb{Y})+\widehat{\xi}^{(n)}(\mathbb{Y})) \big] \\
& \leq \mathbb{P}( \widetilde{\xi}^{(n)}\neq\widehat{\xi}^{(n)} )^{1/2} \big( (\E \widetilde{\xi}^{(n)}(\mathbb{Y})^2)^{1/2} + (\E \widetilde{\xi}^{(n)}(\mathbb{Y})^2)^{1/2} \big) \\
& \leq \varepsilon_n \sqrt{\sum_{\ell=1}^\infty \frac{1}{\ell^2} } \big( (\E \widetilde{\xi}^{(n)}(\mathbb{Y})^2)^{1/2} + (\E \widetilde{\xi}^{(n)}(\mathbb{Y})^2)^{1/2} \big).
\end{align*}
Because of $\E \widetilde{\xi}^{(n)}(\mathbb{Y})^2, \E \widetilde{\xi}^{(n)}(\mathbb{Y})^2 \leq \E \eta(B)^k<\infty$ by the assumption on $g$, we see that
$$
\lim_{n\to\infty} d_{KR}(\widetilde{\xi}^{(n)},\widehat{\xi}^{(n)})=0.
$$
From \cite[Theorem 3.1]{dst} we obtain that
$$
d_{KR}(\widehat{\xi}^{(n)},\zeta) \leq d_{TV}(\widehat{\mathbf{L}}^{(n)},\mathbf{M}) + 2 (\E \widehat{\xi}^{(n)}(\mathbb{Y})^2 - (\E \widehat{\xi}^{(n)}(\mathbb{Y}))^2 - \E \widehat{\xi}^{(n)}(\mathbb{Y}) ),
$$
where $\widehat{\mathbf{L}}^{(n)}$ is the intensity measure of $\widehat{\xi}^{(n)}$. Note that
$$
\widehat{\mathbf{L}}^{(n)} = \mathbf{L}.
$$
Combining this with Lemma \ref{lem:limits_S_n}, which is applicable as $\widehat{\xi}^{(n)}(\mathbb{Y})$ is a $U$-statistic, yields that
$$
\limsup_{n\to\infty} d_{KR}(\widehat{\xi}^{(n)},\zeta) \leq d_{TV}(\mathbf{L},\mathbf{M}) + 2 (\E \widehat{\xi}(\mathbb{Y})^2 - (\E \widehat{\xi}(\mathbb{Y}))^2 - \E \widehat{\xi}(\mathbb{Y}) ),
$$
which is the desired statement under the additional assumption on $\operatorname{dom}'g$.

Next we consider the situation of general $g$. Since $\lambda$ is $\sigma$-finite, there exists an increasing sequence $(B_m)_{m\in\mathbb{N}}$ such that $\lambda(B_m)<\infty$ for all $m\in\mathbb{N}$ and $\bigcup_{m=1}^\infty B_m=\mathbb{X}$. For $m\in\mathbb{N}$ we define
$$
\xi_m=\frac{1}{k!}\sum_{(x_1,\hdots,x_k)\in\eta^k_{\neq}\cap \operatorname{dom}'g \cap B_m^k} \delta_{f(x_1,\hdots,x_k,(M_{i,j})_{i,j\in[k],i\neq j})}
$$
and denote by $\mathbf{L}_m$ the intensity measure of $\xi_m$. From the triangle inequality and the first part of the proof we know that
\begin{align*}
d_{KR}(\xi,\zeta) & \leq d_{KR}(\xi_m,\xi) + d_{KR}(\xi_m,\zeta) \\
& \leq  d_{KR}(\xi_m,\xi) + d_{TV}(\mathbf{L}_m,\mathbf{M}) + 2(\E\xi_m(\mathbb{Y})^2 - (\E\xi_m(\mathbb{Y}))^2 - \E\xi_m(\mathbb{Y}) )
\end{align*}
Because every point of $\xi_m$ is also contained in $\xi$, we deduce that
$$
d_{KR}(\xi_m,\xi) \leq \E[|\xi(\mathbb{Y})-\xi_m(\mathbb{Y})|] = \E[\xi(\mathbb{Y})] - \E[\xi_m(\mathbb{Y})].
$$
and
$$
d_{TV}(\mathbf{L}_m,\mathbf{L}) \leq \E[\xi(\mathbb{Y})] - \E[\xi_m(\mathbb{Y})].
$$
Together with the triangle inequality, the latter leads to
$$
d_{TV}(\mathbf{L}_m,\mathbf{M}) \leq d_{TV}(\mathbf{L}_m,\mathbf{L}) + d_{TV}(\mathbf{L},\mathbf{M}) \leq d_{TV}(\mathbf{L},\mathbf{M})  +  \E[\xi(\mathbb{Y})] - \E[\xi_m(\mathbb{Y})] ,
$$
so that
$$
d_{KR}(\xi,\zeta) \leq d_{TV}(\mathbf{L},\mathbf{M})  + 2 (\E[\xi(\mathbb{Y})] - \E[\xi_m(\mathbb{Y})]) + 2(\E\xi_m(\mathbb{Y})^2 - (\E\xi_m(\mathbb{Y}))^2 - \E\xi_m(\mathbb{Y}) ).
$$
Since, by the dominated convergence theorem, $\E\xi_m(\mathbb{Y})^2\to \E\xi(\mathbb{Y})^2$ and $\E\xi_m(\mathbb{Y})\to \E\xi(\mathbb{Y})$ as $m\to\infty$, we obtain
$$
d_{KR}(\xi,\zeta) \leq d_{TV}(\mathbf{L},\mathbf{M}) + 2(\E\xi(\mathbb{Y})^2 - (\E\xi(\mathbb{Y}))^2 - \E\xi(\mathbb{Y}) ),
$$
which completes the proof.
\end{proof}

Using the above notation, define
 \begin{align}\label{originalUstat}
     S_t:= \frac{1}{k!}\sum_{(x_1, \ldots, x_k) \in \eta^k_{\neq}} f_t \Big( x_1, \ldots, x_k, (W_{x_i,x_j})_{i,j =1,\ldots, k, i \neq j} \Big) ,
 \end{align}
where $f_t$, $t \geq 1$, is a measurable function and the sum is taken over all $k$-tuples of pairwise distinct points. Note that $S$ is not a Poisson $U$-statistics. Now consider the following. For two points $\hat{x}=(x,\mathbf{u}=(u_{i,l})_{i,l\in\mathbb{N}})$ and $\hat{y}=(y,\mathbf{v}=(v_{i,l})_{i,l\in\mathbb{N}})$ in $\hat{\eta}$ such that $x\leq y$ and $y\in A_i$ define $\hat{W}_{\hat{x},\hat{y}} := u_{i,1}$, and
$$ \hat{S}_t:= \frac{1}{k!}\sum_{(\hat{x}_1, \ldots, \hat{x}_k) \in \hat{\eta}^k_{\neq}} f_t \Big( {x}_1, \ldots, {x}_k, (\hat{W}_{\hat{x}_i,\hat{x}_j}) \Big) =:  \sum_{(\hat{x}_1, \ldots, \hat{x}_k) \in \hat{\eta}^k_{\neq}} \hat{f}_t ( \hat{x}_1, \ldots, \hat{x}_k ),$$
which on the other hand is a Poisson U-statistics. Now, choose the partition $\{A_i\}_{i\in\mathbb{N}}$ such that $\lambda_d (A_i) = \frac{\varepsilon}{i}$ for every $i \in \mathbb{N}$, where $\varepsilon>0$ is a given (later chosen to be sufficiently small) number. Then we have
\begin{align}\label{ShatS}
\begin{split}
    \mathbb{P} (S \neq \hat{S}) &= \mathbb{P} ( \exists i \in \mathbb{N} : \eta (A_i) \geq 2 ) \leq \sum_{i=1}^\infty \mathbb{P} ( \eta (A_i) \geq 2 ) \\
    & \leq \sum_{i=1}^\infty  \lambda_d (A_i)^2 = \varepsilon^2 \sum_{i=1}^\infty \frac{1}{k^2},
\end{split}
\end{align}
so that $\mathbb{P} (S \neq \hat{S}) \to 0$, as $\varepsilon \to 0$. Moreover, for every random variable $N$ we have
\begin{align} \label{Kineq}
     d_K (S,N) \leq \mathbb{P} (S \neq \hat{S}) + d_K (\hat{S},N) .
\end{align}

\subsection{Power-weighted edge-lengths}

Consider the statistics

$$L_\rho (\tau):=\frac{1}{2} \sum_{(z_1, z_2) \in \eta^2_{\neq} \cap B^2(0, \rho)} \mathbf{1}\{ z_1 \leftrightarrow z_2 \} \| z_1- z_2 \|^\tau, \quad \tau \in \mathbb{R}.$$

Proceeding as in \cite[Theorem 3.1]{reitznerschultethaele} gives the following formula for its expectation structure. Recall that $g$ denotes the connection function. We can assume that $g$ is only radially dependent, and denote its radial part by $\bar{g}$.

\begin{lemma} \label{lem:expectedlength}
    It holds
    $$ \mathbb{E}[L_\rho (\tau)]= \frac{1}{2} \int_0^\infty \bar{g} (r) r^{\tau + d -1} \int_{\mathbb{S}^{d-1}
} f_\rho (r u) du dr,$$
where
$$f_\rho (ru) = \lambda_d \left( B(0,\rho) \cap (B(0,\rho) + ru) \right)$$
and $du$ stands for the infinitesimal element of the spherical Lebesgue measure. Moreover, it holds
\begin{align*}
    & \frac{1}{2} d \kappa_d \lambda_d(B(0,\rho)) \int_0^\infty \bar{g} (r) r^{\tau + d -1} dr -\kappa_{d-1} S(B(0,\rho)) \int_0^\infty \bar{g} (r) r^{\tau + d -1} dr\\
    & \leq \mathbb{E}[L_\rho (\tau)] \leq \frac{1}{2} d \kappa_d \lambda_d(B(0,\rho)) \int_0^\infty \bar{g} (r) r^{\tau + d -1} dr,
\end{align*}
where $\kappa_d$ denotes the volume of the $d$-dimensional unit ball and $S(B(0,\rho))$ denotes the surface area of the ball  $B(0,\rho)$.
\end{lemma}

Note that Lemma \ref{lem:expectedlength} shows that the expectation of power-weighted edge-lengths is finite if and only if
$$\int_0^\infty \bar{g} (r) r^{\tau + d -1} dr< \infty.$$
If we assume that $\bar{g}$ has polynomial decay $\bar{g}(r) \sim r^{-\alpha}$, $r \to \infty$, for some exponent $\alpha$, this translates to $\alpha>\tau +d$.

Next we give a formula for the variance.


\begin{lemma} \label{lem:varlength}
    Assume that
    $$\int_0^\infty \bar{g} (r) r^{\tau + d -1} dr< \infty.$$
    Then $L_\rho (\tau)$ also has finite variance given by
    \begin{align}\label{varformula}
    \begin{split}
        & \operatorname{Var}[L_\rho (\tau)] = \frac{1}{2} \int_{B(0,\rho)}\int_{B(0,\rho)} g(x-y) \| y-x \|^{2\tau} dx  \\
        & \quad + \int_{B(0,\rho)}\int_{B(0,\rho)} g(x_1-y) \| y-x_1 \|^\tau dx_1 \int_{B(0,\rho)} g(x_2-y)  \| y-x_2 \|^\tau dx_2 dy.
    \end{split} 
    \end{align}
Moreover, if $\bar{g}$ has polynomial decay $\bar{g}(r) \sim r^{-\alpha}$, $r \to \infty$, for some exponent $\alpha$, there is a constant $C \in (0, \infty)$ independent of $\rho$ such that
$$ \frac{1}{C} \rho^{2d+2\tau - \alpha} \leq \operatorname{Var}[L_\rho (\tau)] \leq C\rho^{2d+2\tau }.$$
\end{lemma}

\begin{proof}
    Under the assumption of finite expectations, the formula in \eqref{varformula} can be proven by proceeding as in the proof of \cite[Theorem 3.3]{reitznerschultethaele}. Based on this formula, we now calculate the order of the variance. Using a substitution for the first term on the right-hand side of \eqref{varformula} and the assumption on $\bar{g}$ we have
    \begin{align*}
        &\int_{B(0,\rho)}\int_{B(0,\rho)} g(x-y) \| y-x \|^{2\tau} dx dy\\
        & = \rho^{2d+2\tau} \int_{B(0,1)}\int_{B(0,1)} \bar{g}( \rho\|x-y\| ) \| y-x \|^{2\tau} dxdy \\
        & \geq  \rho^{2d+2\tau} \int_{B(0,1)}\int_{B(0,1)} \mathbf{1}{(\|x-y\| \geq \delta)} \bar{g}( \rho\|x-y\| ) \| y-x \|^{2\tau} dxdy\\
        & \sim \rho^{2d+2\tau-\alpha} \int_{B(0,1)}\int_{B(0,1)} \mathbf{1}{(\|x-y\| \geq \delta)} \|x-y\|^{-\alpha}  \| y-x \|^{2\tau} dxdy .
    \end{align*}
    Now the assertion follows by observing that
    \begin{align*}
         \int_{B(0,\rho)}\int_{B(0,\rho)} g(x_1-y) \| y-x_1 \|^\tau dx_1 \int_{B(0,\rho)} g(x_2-y)  \| y-x_2 \|^\tau dx_2 dy \leq C \rho^{2d+2\tau }.
    \end{align*}
\end{proof}

From the statements in Lemma \ref{lem:varlength} we observe that the variance asymptotics is not finite if $\alpha < 2d+2\tau$.
